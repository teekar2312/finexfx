    17→// Seed-based pseudo-random for consistent mock data
    18→function seededRandom(seed: number) {
    19→  let s = seed;
    20→  return () => {
    21→    s = (s * 16807 + 0) % 2147483647;
    22→    return (s - 1) / 2147483646;
    23→  };
    24→}

    26→function generateMockEquityCurve(days: number) {
    27→  const rng = seededRandom(42);
    28→  const data: { date: string; equity: number; peak: number; drawdown: number }[] = [];
    29→  let equity = 10000;
    30→  let peak = 10000;
    31→  const now = new Date();
    32→  for (let i = days - 1; i >= 0; i--) {
    33→    const d = new Date(now);
    34→    d.setDate(d.getDate() - i);
    35→    const dayReturn = (rng() - 0.42) * 80; // slight positive bias
    36→    equity = Math.max(9200, equity + dayReturn);
    37→    peak = Math.max(peak, equity);
    38→    const dd = ((peak - equity) / peak) * 100;
    39→    data.push({
    40→      date: d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' }),
    41→      equity: Math.round(equity * 100) / 100,
    42→      peak: Math.round(peak * 100) / 100,
    43→      drawdown: Math.round(dd * 100) / 100,
    44→    });
    45→  }
    46→  return data;
    47→}

    49→function generateMockDailyPnl(days: number) {
    50→  const rng = seededRandom(123);
    51→  const data: { date: string; pnl: number }[] = [];
    52→  const now = new Date();
    53→  for (let i = days - 1; i >= 0; i--) {
    54→    const d = new Date(now);
    55→    d.setDate(d.getDate() - i);
    56→    const pnl = (rng() - 0.42) * 120;
    57→    data.push({
    58→      date: d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' }),
    59→      pnl: Math.round(pnl * 100) / 100,
    60→    });
    61→  }
    62→  return data;
    63→}

    65→function generateMockHeatmap(): { day: string; week: number; pnl: number }[] {
    66→  const rng = seededRandom(777);
    67→  const days = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
    68→  const data: { day: string; week: number; pnl: number }[] = [];
    69→  for (let w = 0; w < 5; w++) {
    70→    for (const day of days) {
    71→      if (day === 'Sat' || day === 'Sun') {
    72→        data.push({ day, week: w, pnl: 0 });
    73→      } else {
    74→        const pnl = (rng() - 0.4) * 200;
    75→        data.push({ day, week: w, pnl: Math.round(pnl) });
    76→      }
    77→    }
    78→  }
    79→  return data;
    80→}
