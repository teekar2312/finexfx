    49→    49→// --- Animated Ring Gauge ---
    50→    50→function ConfidenceRing({ confidence, size = 72 }: { confidence: number; size?: number }) {
    51→    51→  const strokeWidth = 5;
    52→    52→  const radius = (size - strokeWidth) / 2;
    53→    53→  const circumference = 2 * Math.PI * radius;
    54→    54→  const progress = (confidence / 100) * circumference;
    55→    55→  const color = confidence >= 80 ? '#10b981' : confidence >= 60 ? '#f59e0b' : '#ef4444';
    56→    56→  const center = size / 2;
    57→    57→
    58→    58→  return (
    59→    59→    <div className="relative" style={{ width: size, height: size }}>
    60→    60→      <svg width={size} height={size} className="-rotate-90">
    61→    61→        <circle
    62→    62→          cx={center}
    63→    63→          cy={center}
    64→    64→          r={radius}
    65→    65→          fill="none"
    66→    66→          stroke="currentColor"
    67→    67→          strokeWidth={strokeWidth}
    68→    68→          className="text-muted/30"
    69→    69→        />
    70→    70→        <motion.circle
    71→    71→          cx={center}
    72→    72→          cy={center}
    73→    73→          r={radius}
    74→    74→          fill="none"
    75→    75→          stroke={color}
    76→    76→          strokeWidth={strokeWidth}
    77→    77→          strokeLinecap="round"
    78→    78→          strokeDasharray={circumference}
    79→    79→          initial={{ strokeDashoffset: circumference }}
    80→    80→          animate={{ strokeDashoffset: circumference - progress }}
    81→    81→          transition={{ duration: 0.8, ease: 'easeOut' }}
    82→    82→        />
    83→    83→      </svg>
    84→    84→      <div className="absolute inset-0 flex items-center justify-center">
    85→    85→        <span className="text-lg font-bold tabular-nums" style={{ color }}>
    86→    86→          {confidence}%
    87→    87→        </span>
    88→    88→      </div>
    89→    89→    </div>
    90→    90→  );
    91→    91→}

    93→    93→// --- Strategy signal generation (deterministic for consistency) ---
    94→    94→function getStrategySignalForSymbol(
    95→    95→  strategy: StrategyName,
    96→    96→  symbol: string,
    97→    97→  mainDirection: TradeDirection | 'HOLD'
    98→    98→): 'bullish' | 'bearish' | 'neutral' {
    99→    99→  const seed = (strategy.length * 7 + symbol.length * 13) % 100;
   100→   100→  if (seed < 55) return mainDirection === 'BUY' ? 'bullish' : mainDirection === 'SELL' ? 'bearish' : 'neutral';
   101→   101→  if (seed < 75) return mainDirection === 'BUY' ? 'bearish' : mainDirection === 'SELL' ? 'bullish' : 'neutral';
   102→   102→  return 'neutral';
   103→   103→}
   104→   104→
   105→   105→function getStrategyConfidence(strategy: StrategyName, symbol: string): number {
   106→   106→  const seed = (strategy.length * 11 + symbol.length * 17) % 100;
   107→   107→  return 40 + seed * 0.5;
   108→   108→}

   110→   110→// --- Indicator data generation (deterministic) ---
   111→   111→function generateIndicatorsForSignal(signal: TradingSignal) {
   112→   112→  const allIndicators = [
   113→   113→    { name: 'RSI(14)', value: 62.4, signal: 'bullish' as const },
   114→   114→    { name: 'MACD(12,26,9)', value: 0.0032, signal: 'bullish' as const },
   115→   115→    { name: 'EMA(9)', value: 1.0845, signal: 'bullish' as const },
   116→   116→    { name: 'EMA(21)', value: 1.0838, signal: 'bullish' as const },
   117→   117→    { name: 'ATR(14)', value: 0.0045, signal: 'neutral' as const },
   118→   118→    { name: 'Bollinger Bands', value: 1.0850, signal: 'neutral' as const },
   119→   119→    { name: 'Stochastic(14,3)', value: 71.2, signal: 'bullish' as const },
   120→   120→    { name: 'SuperTrend(10,3)', value: 1.0830, signal: 'bearish' as const },
   121→   121→    { name: 'Volume', value: 1523, signal: 'bullish' as const },
   122→   122→    { name: 'CCI(20)', value: 85.3, signal: 'bullish' as const },
   123→   123→  ];
   124→   124→
   125→   125→  const seed = (signal.id.charCodeAt(0) * 3 + signal.symbol.length * 7) % allIndicators.length;
   126→   126→  const rotated = [...allIndicators.slice(seed), ...allIndicators.slice(0, seed)];
   127→   127→
   128→   128→  return rotated.slice(0, 8).map((ind) => {
   129→   129→    const isBuy = signal.direction === 'BUY';
   130→   130→    const s = (ind.name.length * 5 + signal.symbol.length * 11) % 10;
   131→   131→    let sig = ind.signal;
   132→   132→    if (s < 2) sig = isBuy ? 'bearish' : 'bullish';
   133→   133→    else if (s < 4) sig = 'neutral';
   134→   134→    return { ...ind, signal: sig };
   135→   135→  });
   136→   136→}
   137→   137→
   138→   138→// --- Historical accuracy simulation ---
   139→   139→function generateHistoricalAccuracy(signal: TradingSignal) {
   140→   140→  const seed = (signal.symbol.length * 13 + signal.strategy.length * 7) % 100;
   141→   141→  const winRate = 55 + (seed % 25);
   142→   142→  const lastSignals = Array.from({ length: 10 }, (_, i) => {
   143→   143→    const s = (seed + i * 17) % 10;
   144→   144→    return s < winRate / 10 ? 'win' : 'loss';
   145→   145→  });
   146→   146→  const avgPnl = (winRate - 50) * 1.8 + (seed % 5) - 2;
   147→   147→  const bestTrade = 20 + seed % 60;
   148→   148→  const worstTrade = -(10 + seed % 25);
   149→   149→  return { winRate, lastSignals, avgPnl, bestTrade, worstTrade };
   150→   150→}
