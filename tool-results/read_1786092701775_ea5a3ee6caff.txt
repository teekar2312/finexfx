   127→    const potentialProfit = tpPips * pipValue;

   130→      symbol: SYMBOL_INFO[selectedSymbol].name,
   131→      direction,
   132→      lotSize: lots,
   133→      entryPrice,
   134→      stopLoss: slPrice,
   135→      takeProfit: tpPrice,
   136→      riskAmount,
   137→      potentialProfit,
   138→      riskRewardRatio: rrRatio,
   139→      spread: parseFloat((price.spread / pipSize).toFixed(1)),
   140→      commission: BROKER_CONFIG.commission * lots,
   141→      accountBalance: balance,
   142→      marginRequired: margin,
   143→      freeMargin: balance - margin,

   164→    const tpPips = parseFloat(takeProfit) || 0;

   171→    const margin = (contractSize * lots * currentPrice) / BROKER_CONFIG.leverage;

   174→    const potentialProfit = tpPips * pipValue;

   198→    const min = diffMin % 60;
   199→    if (hr < 24) return `${hr}h ${min}m`;
   200→    const days = Math.floor(hr / 24);
   201→    const remHr = hr % 24;

   338→                                <TableCell className="text-xs font-medium">{trade.symbol}</TableCell>
   339→                                <TableCell>
   340→                                  <div className={`flex items-center gap-1 text-xs font-bold ${trade.direction === 'BUY' ? 'text-emerald-500' : 'text-red-500'}`}>

   395→                                        initial={{ opacity: 0, scale: 0.8 }}
   396→                                        animate={{ opacity: 1, scale: 1 }}

   424→                                        <Square className="h-3 w-3" />
   425→                                      </Button>
