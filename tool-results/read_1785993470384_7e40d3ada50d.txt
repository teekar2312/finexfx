    23→    23→function getSessionStatus(session: { start: number; end: number }) {
    24→    24→  const now = new Date();
    25→    25→  const currentHour = now.getUTCHours();
    26→    26→  const currentMinute = now.getUTCMinutes();
    27→    27→  const currentDecimal = currentHour + currentMinute / 60;
    28→    28→  const isActive = currentDecimal >= session.start && currentDecimal < session.end;
    29→    29→  const progress = isActive
    30→    30→    ? ((currentDecimal - session.start) / (session.end - session.start)) * 100
    31→    31→    : 0;
    32→    32→  const duration = session.end - session.start;
    33→    33→  let statusText = '';
    34→    34→  if (isActive) {
    35→    35→    statusText = 'ACTIVE';
    36→    36→  } else if (currentDecimal < session.start) {
    37→    37→    const hoursUntil = session.start - currentHour;
    38→    38→    const minsUntil = (hoursUntil * 60 - currentMinute);
    39→    39→    if (minsUntil > 0) {
    40→    40→      const h = Math.floor(minsUntil / 60);
    41→    41→      const m = minsUntil % 60;
    42→    42→      statusText = `Opens in ${h}:${m.toString().padStart(2, '0')}`;
    43→    43→    } else {
    44→    44→      statusText = 'Closed';
    45→    45→    }
    46→    46→  } else {
    47→    47→    statusText = 'Closed';
    48→    48→  }
    49→    49→  return { isActive, progress, statusText, duration };
    50→    50→}

    52→    52→function timeAgo(dateStr: string): string {
    53→    53→  const now = Date.now();
    54→    54→  const then = new Date(dateStr).getTime();
    55→    55→  const diffMs = now - then;
    56→    56→  const diffSec = Math.floor(diffMs / 1000);
    57→    57→  if (diffSec < 60) return `${diffSec}s ago`;
    58→    58→  const diffMin = Math.floor(diffSec / 60);
    59→    59→  if (diffMin < 60) return `${diffMin}m ago`;
    60→    60→  const diffHr = Math.floor(diffMin / 60);
    61→    61→  return `${diffHr}h ago`;
    62→    62→}

    64→    64→function MiniSparkline({ values, width = 48, height = 18 }: { values: number[]; width?: number; height?: number }) {
    65→    65→  if (values.length < 2) return null;
    66→    66→  const min = Math.min(...values);
    67→    67→  const max = Math.max(...values);
    68→    68→  const range = max - min || 1;
    69→    69→  const padding = 2;
    70→    70→  const points = values.map((v, i) => {
    71→    71→    const x = padding + (i / (values.length - 1)) * (width - padding * 2);
    72→    72→    const y = padding + (1 - (v - min) / range) * (height - padding * 2);
    73→    73→    return `${x},${y}`;
    74→    74→  }).join(' ');
    75→    75→  const isUp = values[values.length - 1] >= values[0];
    76→    76→  const color = isUp ? '#10b981' : '#ef4444';
    77→    77→  return (
    78→    78→    <svg width={width} height={height} className="inline-block">
    79→    79→      <polyline
    80→    80→        fill="none"
    81→    81→        stroke={color}
    82→    82→        strokeWidth={1.5}
    83→    83→        strokeLinecap="round"
    84→    84→        strokeLinejoin="round"
    85→    85→        points={points}
    86→    86→      />
    87→    87→    </svg>
    88→    88→  );
    89→    89→}
