    60→function formatDuration(ms: number): string {
    61→  const totalSec = Math.floor(ms / 1000);
    62→  const h = Math.floor(totalSec / 3600);
    63→  const m = Math.floor((totalSec % 3600) / 60);
    64→  const s = totalSec % 60;
    65→  if (h > 0) return `${h}h ${m}m`;
    66→  if (m > 0) return `${m}m ${s}s`;
    67→  return `${s}s`;
    68→}
    69→
    70→function formatDateTime(iso: string): string {
    71→  const d = new Date(iso);
    72→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' }) +
    73→    ' ' +
    74→    d.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', hour12: false });
    75→}
    76→
    77→function priceDigits(symbol: Symbol): number {
    78→  return SYMBOL_INFO[symbol].digits;
    79→}

   122→function TradeHistoryTable() {
   123→  const storeClosedTrades = useTradingStore((s) => s.closedTrades);
   124→
   125→  // Normalise store trades or use mock fallback
   126→  const trades: ClosedTradeRow[] = useMemo(() => {
   127→    if (storeClosedTrades.length > 0) {
   128→      return storeClosedTrades.map((t) => ({
   129→        id: t.id,
   130→        openedAt: t.openedAt,
   131→        closedAt: t.closedAt ?? new Date().toISOString(),
   132→        symbol: t.symbol,
   133→        direction: t.direction,
   134→        lotSize: t.lotSize,
   135→        entryPrice: t.entryPrice,
   136→        exitPrice: t.currentPrice,
   137→        stopLoss: t.stopLoss ?? 0,
   138→        takeProfit: t.takeProfit ?? 0,
   139→        profit: t.profit,
   140→        pips: t.pips,
   141→        duration: t.closedAt
   142→          ? new Date(t.closedAt).getTime() - new Date(t.openedAt).getTime()
   143→          : Date.now() - new Date(t.openedAt).getTime(),
   144→        strategy: t.strategy ?? '—',
   145→      }));
   146→    }
   147→    return generateMockTrades();
   148→  }, [storeClosedTrades]);
   149→
   150→  // ---- Filters ----
   151→  const [search, setSearch] = useState('');
   152→  const [pairFilter, setPairFilter] = useState<PairFilter>('ALL');
   153→  const [typeFilter, setTypeFilter] = useState<TypeFilter>('ALL');
   154→  const [resultFilter, setResultFilter] = useState<ResultFilter>('ALL');
   155→  const [dateRange, setDateRange] = useState<DateRange>('ALL');
   156→
   157→  // ---- Sort ----
   158→  const [sortKey, setSortKey] = useState<SortKey>('openedAt');
   159→  const [sortDir, setSortDir] = useState<SortDir>('desc');
   160→
   161→  const toggleSort = useCallback((key: SortKey) => {
   162→    setSortDir((prev) => {
   163→      if (sortKey === key) return prev === 'asc' ? 'desc' : 'asc';
   164→      return 'asc';
   165→    });
   166→    setSortKey(key);
   167→  }, [sortKey]);
   168→
   169→  // ---- Pagination ----
   170→  const [page, setPage] = useState(1);
   171→  const perPage = 10;
   172→
   173→  // ---- Filtering & sorting ----
   174→  const now = Date.now();
   175→
   176→  const filtered = useMemo(() => {
   177→    let list = [...trades];
   178→
   179→    // Search
   180→    if (search.trim()) {
   181→      const q = search.toLowerCase();
   182→      list = list.filter(
   183→        (t) =>
   184→          t.id.toLowerCase().includes(q) ||
   185→          t.symbol.toLowerCase().includes(q)
   186→      );
   187→    }
   188→
   189→    // Pair
   190→    if (pairFilter !== 'ALL') list = list.filter((t) => t.symbol === pairFilter);
   191→
   192→    // Type
   193→    if (typeFilter !== 'ALL') list = list.filter((t) => t.direction === typeFilter);
   194→
   195→    // Result
   196→    if (resultFilter === 'Profit') list = list.filter((t) => t.profit > 0);
   197→    else if (resultFilter === 'Loss') list = list.filter((t) => t.profit < 0);
   198→    else if (resultFilter === 'Breakeven') list = list.filter((t) => t.profit === 0);
   199→
   200→    // Date range
   201→    if (dateRange !== 'ALL') {
   202→      const cutoff = dateRange === 'Today' ? now - 24 * 3600000
   203→        : dateRange === 'This Week' ? now - 7 * 86400000
   204→          : now - 30 * 86400000;
   205→      list = list.filter((t) => new Date(t.closedAt).getTime() >= cutoff);
   206→    }
   207→
   208→    // Sort
   209→    list.sort((a, b) => {
   210→      let va: string | number = a[sortKey];
   211→      let vb: string | number = b[sortKey];
   212→      if (typeof va === 'string') va = va.toLowerCase();
   213→      if (typeof vb === 'string') vb = vb.toLowerCase();
   214→      if (va < vb) return sortDir === 'asc' ? -1 : 1;
   215→      if (va > vb) return sortDir === 'asc' ? 1 : -1;
   216→      return 0;
   217→    });
   218→
   219→    return list;
   220→  }, [trades, search, pairFilter, typeFilter, resultFilter, dateRange, sortKey, sortDir, now]);
   221→
   222→  // Reset page when filters change
   223→  const totalPages = Math.max(1, Math.ceil(filtered.length / perPage));
   224→  const safePage = Math.min(page, totalPages);
   225→  const paginated = filtered.slice((safePage - 1) * perPage, safePage * perPage);
   226→
   227→  // ---- Summary stats ----
   228→  const stats = useMemo(() => {
   229→    const wins = filtered.filter((t) => t.profit > 0);
   230→    const losses = filtered.filter((t) => t.profit < 0);
   231→    const totalPnl = filtered.reduce((s, t) => s + t.profit, 0);
   232→    const avgWin = wins.length ? wins.reduce((s, t) => s + t.profit, 0) / wins.length : 0;
   233→    const avgLoss = losses.length ? losses.reduce((s, t) => s + t.profit, 0) / losses.length : 0;
   234→    const best = filtered.length ? Math.max(...filtered.map((t) => t.profit)) : 0;
   235→    const worst = filtered.length ? Math.min(...filtered.map((t) => t.profit)) : 0;
   236→    const avgDuration = filtered.length
   237→      ? filtered.reduce((s, t) => s + t.duration, 0) / filtered.length
   238→      : 0;
   239→    const winRate = filtered.length ? (wins.length / filtered.length) * 100 : 0;
   240→    return { total: filtered.length, winRate, totalPnl, avgWin, avgLoss, best, worst, avgDuration };
   241→  }, [filtered]);
   242→
   243→  /* ---- Sort header helper ---- */
   244→  const SortIcon = ({ col }: { col: SortKey }) => {

   246→    return sortDir === 'asc'
   247→      ? <ArrowUp className="w-3 h-3 ml-1 text-emerald-400" />
   248→      : <ArrowDown className="w-3 h-3 ml-1 text-emerald-400" />;
   249→  };

   269→      }`}
   270→    >
   271→      {label}

   379→                <th className="px-3 py-2.5 text-right text-[10px] uppercase tracking-wider text-muted-foreground font-medium whitespace-nowrap cursor-pointer select-none hover:text-foreground transition-colors" onClick={() => toggleSort('pips')}>
   380→                  <span className="inline-flex items-center justify-end">P&L (pips) <SortIcon col="pips" /></span>
   381→                </th>
   382→                <th className="px-3 py-2.5 text-right text-[10px] uppercase tracking-wider text-muted-foreground font-medium whitespace-nowrap cursor-pointer select-none hover:text-foreground transition-colors" onClick={() => toggleSort('duration')}>
   383→                  <span className="inline-flex items-center justify-end">Duration <SortIcon col="duration" /></span>
   384→                </th>
   385→                <th className="px-3 py-2.5 text-left text-[10px] uppercase tracking-wider text-muted-foreground font-medium whitespace-nowrap">Strategy</th>
   386→              </tr>
   387→            </thead>
   388→
   389→            <tbody>
   390→              <AnimatePresence mode="popLayout">
   391→                {paginated.map((trade, idx) => (
   392→                  <motion.tr
   393→                    key={trade.id}
   394→                    initial={{ opacity: 0, y: 8 }}
   395→                    animate={{ opacity: 1, y: 0 }}
   396→                    exit={{ opacity: 0, y: -8 }}
   397→                    transition={{ duration: 0.25, delay: idx * 0.03 }}
   398→                    className={`border-b border-white/5 hover:bg-white/[0.03] transition-colors ${
   399→                      trade.profit > 0
   400→                        ? 'border-l-2 border-l-emerald-500/60'
   401→                        : trade.profit < 0
   402→                          ? 'border-l-2 border-l-red-500/60'
   403→                          : 'border-l-2 border-l-white/10'
   404→                    }`}
   405→                  >
   406→                    <td className="px-3 py-2 font-mono text-muted-foreground whitespace-nowrap">{trade.id}</td>
   407→                    <td className="px-3 py-2 whitespace-nowrap">
   408→                      <div className="text-foreground">{formatDateTime(trade.openedAt)}</div>
   409→                      <div className="text-muted-foreground text-[10px]">{formatDateTime(trade.closedAt)}</div>
   410→                    </td>
   411→                    <td className="px-3 py-2 font-medium whitespace-nowrap">{SYMBOL_INFO[trade.symbol].name}</td>
   412→                    <td className={`px-3 py-2 font-semibold whitespace-nowrap ${trade.direction === 'BUY' ? 'text-emerald-400' : 'text-red-400'}`}>
   413→                      {trade.direction}
   414→                    </td>
   415→                    <td className="px-3 py-2 text-right font-mono whitespace-nowrap">{trade.lotSize.toFixed(2)}</td>
   416→                    <td className="px-3 py-2 text-right font-mono whitespace-nowrap">{trade.entryPrice.toFixed(priceDigits(trade.symbol))}</td>
   417→                    <td className="px-3 py-2 text-right font-mono whitespace-nowrap">{trade.exitPrice.toFixed(priceDigits(trade.symbol))}</td>
   418→                    <td className="px-3 py-2 text-right font-mono text-muted-foreground whitespace-nowrap">{trade.stopLoss > 0 ? trade.stopLoss.toFixed(priceDigits(trade.symbol)) : '—'}</td>
   419→                    <td className="px-3 py-2 text-right font-mono text-muted-foreground whitespace-nowrap">{trade.takeProfit > 0 ? trade.takeProfit.toFixed(priceDigits(trade.symbol)) : '—'}</td>
   420→                    <td className={`px-3 py-2 text-right font-mono font-semibold whitespace-nowrap ${trade.profit > 0 ? 'text-emerald-400' : trade.profit < 0 ? 'text-red-400' : 'text-muted-foreground'}`}>
   421→                      {trade.profit > 0 ? '+' : ''}{trade.profit.toFixed(2)}
   422→                    </td>
   423→                    <td className={`px-3 py-2 text-right font-mono font-semibold whitespace-nowrap ${trade.pips > 0 ? 'text-emerald-400' : trade.pips < 0 ? 'text-red-400' : 'text-muted-foreground'}`}>
   424→                      {trade.pips > 0 ? '+' : ''}{trade.pips.toFixed(1)}
   425→                    </td>
   426→                    <td className="px-3 py-2 text-right font-mono whitespace-nowrap text-muted-foreground">
   427→                      <span className="inline-flex items-center gap-1">
   428→                        <Timer className="w-3 h-3" />
   429→                        {formatDuration(trade.duration)}
   430→                      </span>
   431→                    </td>
   432→                    <td className="px-3 py-2 whitespace-nowrap">
   433→                      <span className="inline-block px-1.5 py-0.5 rounded bg-white/5 text-[10px] text-muted-foreground">{trade.strategy}</span>
   434→                    </td>
   435→                  </motion.tr>
   436→                ))}
   437→              </AnimatePresence>
   438→            </tbody>
   439→          </table>
   440→
   441→          {filtered.length === 0 && (
   442→            <div className="py-12 text-center text-muted-foreground text-sm">
   443→              No trades match the current filters.
   444→            </div>
   445→          )}
   446→        </div>
   447→
   448→        {/* ========== Pagination ========== */}
   449→        {filtered.length > perPage && (
   450→          <div className="flex items-center justify-between px-4 py-2.5 border-t border-white/10">
   451→            <span className="text-xs text-muted-foreground">
   452→              Showing {(safePage - 1) * perPage + 1}–{Math.min(safePage * perPage, filtered.length)} of {filtered.length}
   453→            </span>
   454→            <div className="flex items-center gap-1">
   455→              <Button
   456→                variant="ghost"
   457→                size="sm"
   458→                className="h-7 px-2 text-xs"
   459→                disabled={safePage <= 1}
   460→                onClick={() => setPage((p) => Math.max(1, p - 1))}
   461→              >
   462→                <ChevronLeft className="w-3.5 h-3.5" />
   463→                Prev
   464→              </Button>
   465→              <div className="flex items-center gap-0.5 mx-1">
   466→                {Array.from({ length: totalPages }, (_, i) => i + 1).map((p) => (
   467→                  <button
   468→                    key={p}
   469→                    onClick={() => setPage(p)}
   470→                    className={`w-7 h-7 rounded text-xs font-medium transition-colors ${
   471→                      p === safePage
   472→                        ? 'bg-emerald-500/20 text-emerald-400'
   473→                        : 'text-muted-foreground hover:bg-white/5 hover:text-foreground'
   474→                    }`}
   475→                  >
   476→                    {p}
   477→                  </button>
   478→                ))}
   479→              </div>
   480→              <Button
   481→                variant="ghost"
   482→                size="sm"
   483→                className="h-7 px-2 text-xs"
   484→                disabled={safePage >= totalPages}
   485→                onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   486→              >
   487→                Next
   488→                <ChevronRight className="w-3.5 h-3.5" />
   489→              </Button>
   490→            </div>
   491→          </div>
   492→        )}
   493→      </div>
   494→    </div>
   495→  );
   496→}
   497→
   498→export default TradeHistoryTable;
