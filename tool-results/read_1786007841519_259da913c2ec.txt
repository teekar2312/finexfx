    24→export default function TradingView() {
    25→  const {

    27→    openTrades, closedTrades, addTrade, closeTrade, updateTrade,
    28→    isConnected, riskSettings, addNotification,

    33→  const [takeProfit, setTakeProfit] = useState(riskSettings.takeProfitPips.toString());
    34→  const [oneClickMode, setOneClickMode] = useState(false);
    35→  const [confirmClose, setConfirmClose] = useState<string | null>(null);
    36→  const prevPriceRef = useRef<Record<string, number>>({});
    37→  const [priceFlash, setPriceFlash] = useState<Record<string, 'up' | 'down' | null>>({});
    38→
    39→  const price = prices[selectedSymbol];
    40→  const chartData = priceHistory[selectedSymbol] || [];
    41→
    42→  useEffect(() => {
    43→    if (price) {
    44→      const prev = prevPriceRef.current[selectedSymbol];
    45→      if (prev !== undefined && price.bid !== prev) {
    46→        setPriceFlash(state => ({ ...state, [selectedSymbol]: price.bid > prev ? 'up' : 'down' }));
    47→        const timer = setTimeout(() => {
    48→          setPriceFlash(state => ({ ...state, [selectedSymbol]: null }));
    49→        }, 500);
    50→        return () => clearTimeout(timer);
    51→      }
    52→      prevPriceRef.current[selectedSymbol] = price.bid;
    53→    }
    54→  }, [price, selectedSymbol]);
    55→
    56→  const handleOpenTrade = useCallback((direction: TradeDirection) => {
    57→    if (!price || !isConnected) {
    58→      addNotification({ type: 'error', title: 'Cannot Trade', message: 'Not connected to price feed' });
    59→      return;
    60→    }
    61→

    63→    if (isNaN(lots) || lots < BROKER_CONFIG.minLotSize || lots > BROKER_CONFIG.maxLotSize) {
    64→      addNotification({ type: 'error', title: 'Invalid Lot Size', message: `Must be between ${BROKER_CONFIG.minLotSize} and ${BROKER_CONFIG.maxLotSize}` });
    65→      return;
    66→    }
    67→
    68→    const entryPrice = direction === 'BUY' ? price.ask : price.bid;
    69→    const slPips = parseFloat(stopLoss) || 0;
    70→    const tpPips = parseFloat(takeProfit) || 0;
    71→    const pipSize = SYMBOL_INFO[selectedSymbol].pipSize;
    72→
    73→    const trade = {
    74→      id: `trade-${Date.now()}-${Math.random().toString(36).substr(2, 6)}`,
    75→      symbol: selectedSymbol,
    76→      direction,
    77→      lotSize: lots,
    78→      entryPrice,
    79→      currentPrice: entryPrice,
    80→      stopLoss: slPips > 0 ? (direction === 'BUY' ? entryPrice - slPips * pipSize : entryPrice + slPips * pipSize) : undefined,
    81→      takeProfit: tpPips > 0 ? (direction === 'BUY' ? entryPrice + tpPips * pipSize : entryPrice - tpPips * pipSize) : undefined,
    82→      isTrailingStop: false,
    83→      trailingStop: undefined,
    84→      pips: 0,
    85→      profit: 0,
    86→      commission: BROKER_CONFIG.commission * lots,
    87→      spread: price.spread,
    88→      swap: 0,
    89→      status: 'open' as const,
    90→      openedAt: new Date().toISOString(),
    91→    };
    92→
    93→    addTrade(trade);
    94→  }, [price, isConnected, lotSize, stopLoss, takeProfit, selectedSymbol, addTrade, addNotification]);
    95→
    96→  const handleToggleTrailing = (tradeId: string, enabled: boolean) => {
    97→    updateTrade(tradeId, { isTrailingStop: enabled, trailingStop: enabled ? riskSettings.stopLossPips : undefined });
    98→  };
    99→
   100→  // (c) Calculated order metrics
   101→  const orderMetrics = useMemo(() => {
   102→    const lots = parseFloat(lotSize) || 0;
   103→    const slPips = parseFloat(stopLoss) || 0;
   104→    const tpPips = parseFloat(takeProfit) || 0;
   105→    const pipValue = SYMBOL_INFO[selectedSymbol].category === 'forex' ? 10 * lots : 100 * lots;
   106→    const pipSize = SYMBOL_INFO[selectedSymbol].pipSize;
   107→
   108→    // Margin = (contractSize * lots * price) / leverage
   109→    const contractSize = SYMBOL_INFO[selectedSymbol].category === 'forex' ? 100000 : 100;
   110→    const currentPrice = price ? (selectedSymbol === 'USDJPY' ? 1 / price.bid : price.bid) : 1;
   111→    const margin = (contractSize * lots * currentPrice) / BROKER_CONFIG.leverage;
   112→
   113→    const riskAmount = slPips * pipValue;
   114→    const potentialProfit = tpPips * pipValue;
   115→
   116→    return { margin, riskAmount, potentialProfit, pipValue };
   117→  }, [lotSize, stopLoss, takeProfit, selectedSymbol, price]);
   118→

   120→  const historyStats = useMemo(() => {
   121→    if (closedTrades.length === 0) return null;
   122→    const wins = closedTrades.filter(t => t.profit > 0);
   123→    const losses = closedTrades.filter(t => t.profit <= 0);
   124→    const winRate = (wins.length / closedTrades.length) * 100;
   125→    const totalPnl = closedTrades.reduce((s, t) => s + t.profit, 0);
   126→    const bestTrade = closedTrades.reduce((best, t) => t.profit > best.profit ? t : best, closedTrades[0]);
   127→    const worstTrade = closedTrades.reduce((worst, t) => t.profit < worst.profit ? t : worst, closedTrades[0]);
   128→    return { totalTrades: closedTrades.length, winRate, totalPnl, bestTrade, worstTrade };
   129→  }, [closedTrades]);
   130→
   131→  function formatDuration(openedAt: string, closedAt?: string) {
   132→    const start = new Date(openedAt).getTime();
   133→    const end = (closedAt ? new Date(closedAt) : new Date()).getTime();
   134→    const diffMin = Math.floor((end - start) / 60000);
   135→    if (diffMin < 1) return '<1m';
   136→    if (diffMin < 60) return `${diffMin}m`;
   137→    const hr = Math.floor(diffMin / 60);
   138→    const min = diffMin % 60;
   139→    if (hr < 24) return `${hr}h ${min}m`;
   140→    const days = Math.floor(hr / 24);
   141→    const remHr = hr % 24;
   142→    return `${days}d ${remHr}h`;
   143→  }

   148→    if (!history || history.length < 2) return '';

   150→    const min = Math.min(...pts.map(p => p.bid));
   151→    const max = Math.max(...pts.map(p => p.bid));
   152→    const range = max - min || 1;
   153→    return pts.map((p, i) => `${(i / 11) * 36},${20 - ((p.bid - min) / range) * 18}`).join(' ');
   154→  }, [priceHistory]);
   155→
   156→  // Quick lot sizes
   157→  const quickLots = [0.01, 0.05, 0.1, 0.5, 1.0];
   158→
   159→  return (
   160→    <TooltipProvider delayDuration={0}>
   161→      <div className="p-4 pb-10 md:pb-4 space-y-4">
   162→        {/* Symbol Selector - Pill style with sparklines */}
   163→        <div className="flex gap-2 overflow-x-auto pb-2 -mx-1 px-1">
   164→          {SYMBOLS.map((sym) => {
   165→            const isActive = sym === selectedSymbol;
   166→            const p = prices[sym];
   167→            const sparkline = getSparklinePoints(sym);
   168→            return (
   169→              <button
   170→                key={sym}
   171→                onClick={() => setSelectedSymbol(sym)}
   172→                className={`flex-shrink-0 py-2.5 px-3 rounded-full text-sm font-medium transition-all min-w-[120px] card-hover scale-click flex items-center gap-2 border ${
   173→                  isActive
   174→                    ? 'bg-emerald-500/10 text-emerald-400 border-emerald-500/30 pill-active-glow shadow-[0_0_20px_rgba(16,185,129,0.1)]'
   175→                    : 'bg-card/80 border-border/60 text-muted-foreground hover:text-foreground hover:border-border'
   176→                }`}
   177→              >
   178→                <div className="flex flex-col items-start gap-0.5">
   179→                  <div className="flex items-center gap-1.5">
   180→                    <span className="text-xs font-bold tracking-tight">{SYMBOL_INFO[sym].name}</span>
   181→                    {p && (
   182→                      <span className="text-[9px] text-muted-foreground font-normal tabular-nums">
   183→                        {p.spread.toFixed(1)}p
   184→                      </span>
   185→                    )}
   186→                  </div>
   187→                  {p && (
   188→                    <div className={`text-[11px] tabular-nums font-medium ${p.change >= 0 ? 'text-emerald-500' : 'text-red-500'}`}>
   189→                      {p.bid.toFixed(SYMBOL_INFO[sym].digits)}
   190→                    </div>
   191→                  )}
   192→                </div>
   193→                {sparkline && (
   194→                  <svg viewBox="0 0 36 20" className="w-9 h-5 flex-shrink-0" preserveAspectRatio="none">
   195→                    <polyline
   196→                      points={sparkline}
   197→                      fill="none"
   198→                      stroke={p && p.change >= 0 ? '#10b981' : '#ef4444'}
   199→                      strokeWidth="1.5"
   200→                      strokeLinecap="round"
   201→                      strokeLinejoin="round"
   202→                      opacity={isActive ? 1 : 0.6}
   203→                    />
   204→                  </svg>
   205→                )}
   206→              </button>
   207→            );
   208→          })}
   209→        </div>
   210→
   211→        <div className="grid grid-cols-1 lg:grid-cols-4 gap-4">
   212→          {/* Chart */}
   213→          <div className="lg:col-span-3 space-y-4">
   214→            <div className="chart-glow rounded-xl">
   215→              <Card className="glass-card border-border/40">
   216→                <CardContent className="p-0">
   217→                  <PriceChart
   218→                    data={chartData}
   219→                    symbol={selectedSymbol}
   220→                    bid={price?.bid}
   221→                    ask={price?.ask}
   222→                    height={300}
   223→                  />
   224→                </CardContent>
   225→              </Card>
   226→            </div>
   227→
   228→            {/* (a) Tabbed Positions / History */}
   229→            <Card className="glass-card card-hover">
   230→              <Tabs defaultValue="open" className="w-full">
   231→                <CardHeader className="pb-0 pt-3 px-4">
   232→                  <div className="flex items-center justify-between gap-2">
   233→                    <TabsList className="h-8 bg-slate-800/50">
   234→                      <TabsTrigger value="open" className="text-xs h-6 data-[state=active]:bg-emerald-600/20 data-[state=active]:text-emerald-400 data-[state=active]:shadow-[0_0_10px_rgba(16,185,129,0.15)]">
   235→                        Open ({openTrades.length})
   236→                      </TabsTrigger>
   237→                      <TabsTrigger value="history" className="text-xs h-6 data-[state=active]:bg-emerald-600/20 data-[state=active]:text-emerald-400 data-[state=active]:shadow-[0_0_10px_rgba(16,185,129,0.15)]">
   238→                        History ({closedTrades.length})
   239→                      </TabsTrigger>
   240→                      <TabsTrigger value="advanced" className="text-xs h-6 data-[state=active]:bg-amber-600/20 data-[state=active]:text-amber-400 data-[state=active]:shadow-[0_0_10px_rgba(245,158,11,0.15)]">
   241→                        Advanced
   242→                      </TabsTrigger>
   243→                    </TabsList>
   244→                    <TradeExportButton />
   245→                  </div>
   246→                </CardHeader>
   247→
   248→                <TabsContent value="open" className="mt-0">
   249→                  <CardContent className="px-4 pb-3">
   250→                    {openTrades.length === 0 ? (
   251→                      <div className="text-center py-6 text-muted-foreground text-sm">
   252→                        No open positions. Use the panel to place a trade.
   253→                      </div>
   254→                    ) : (
   255→                      <div className="overflow-x-auto">
   256→                        <Table>
   257→                          <TableHeader>
   258→                            <TableRow className="border-border hover:bg-transparent">
   259→                              <TableHead className="text-[10px] h-8">Symbol</TableHead>
   260→                              <TableHead className="text-[10px] h-8">Dir</TableHead>
   261→                              <TableHead className="text-[10px] h-8 text-right">Lots</TableHead>
   262→                              <TableHead className="text-[10px] h-8 text-right">Entry</TableHead>
   263→                              <TableHead className="text-[10px] h-8 text-right">SL</TableHead>
   264→                              <TableHead className="text-[10px] h-8 text-right">TP</TableHead>
   265→                              <TableHead className="text-[10px] h-8 text-right">Pips</TableHead>
   266→                              <TableHead className="text-[10px] h-8 text-right">P&L</TableHead>
   267→                              <TableHead className="text-[10px] h-8">Time</TableHead>
   268→                              <TableHead className="text-[10px] h-8 text-center">Trail</TableHead>
   269→                              <TableHead className="text-[10px] h-8 text-center">Action</TableHead>
   270→                            </TableRow>
   271→                          </TableHeader>
   272→                          <TableBody>
   273→                            {openTrades.map((trade) => {
   274→                              const maxProfit = Math.max(...openTrades.map(t => Math.abs(t.profit)), 1);
   275→                              return (
   276→                              <TableRow key={trade.id} className={`border-border compact-row ${trade.direction === 'BUY' ? 'trade-row-buy' : 'trade-row-sell'}`}>
   277→                                <TableCell className="text-xs font-medium">{trade.symbol}</TableCell>
   278→                                <TableCell>
   279→                                  <div className={`flex items-center gap-1 text-xs font-bold ${trade.direction === 'BUY' ? 'text-emerald-500' : 'text-red-500'}`}>
   280→                                    {trade.direction === 'BUY' ? <ArrowUpRight className="h-3 w-3" /> : <ArrowDownRight className="h-3 w-3" />}
   281→                                    {trade.direction}
   282→                                  </div>
   283→                                </TableCell>
   284→                                <TableCell className="text-xs text-right tabular-nums">{trade.lotSize}</TableCell>
   285→                                <TableCell className="text-xs text-right tabular-nums">{trade.entryPrice.toFixed(SYMBOL_INFO[trade.symbol].digits)}</TableCell>
   286→                                <TableCell className="text-xs text-right tabular-nums text-red-400">
   287→                                  {trade.stopLoss?.toFixed(SYMBOL_INFO[trade.symbol].digits) || '—'}
   288→                                </TableCell>
   289→                                <TableCell className="text-xs text-right tabular-nums text-emerald-400">
   290→                                  {trade.takeProfit?.toFixed(SYMBOL_INFO[trade.symbol].digits) || '—'}
   291→                                </TableCell>
   292→                                <TableCell className={`text-xs text-right tabular-nums font-semibold ${trade.pips >= 0 ? 'text-emerald-500' : 'text-red-500'}`}>
   293→                                  {trade.pips >= 0 ? '+' : ''}{trade.pips.toFixed(1)}
   294→                                </TableCell>
   295→                                <TableCell className="text-right">
   296→                                  <div className={`text-sm font-bold tabular-nums ${trade.profit >= 0 ? 'gradient-text-profit' : 'gradient-text-loss'}`}>
   297→                                    {trade.profit >= 0 ? '+' : ''}${trade.profit.toFixed(2)}
   298→                                  </div>
   299→                                  <div className="pnl-bar-track mt-0.5">
   300→                                    <div
   301→                                      className={`pnl-bar-fill ${trade.profit >= 0 ? 'bg-emerald-500' : 'bg-red-500'}`}
   302→                                      style={{ width: `${Math.min((Math.abs(trade.profit) / maxProfit) * 100, 100)}%` }}
   303→                                    />
   304→                                  </div>
   305→                                </TableCell>
   306→                                <TableCell className="text-[10px] text-muted-foreground tabular-nums whitespace-nowrap">
   307→                                  <div className="flex items-center gap-1">
   308→                                    <Timer className="h-2.5 w-2.5 opacity-50" />
   309→                                    {formatDuration(trade.openedAt)}
   310→                                  </div>
   311→                                </TableCell>
   312→                                <TableCell className="text-center">
   313→                                  {trade.isTrailingStop ? (
   314→                                    <Tooltip>
   315→                                      <TooltipTrigger>
   316→                                        <div className="trail-animate inline-flex">
   317→                                          <Activity className="h-3.5 w-3.5 text-emerald-500" />
   318→                                        </div>
   319→                                      </TooltipTrigger>
   320→                                      <TooltipContent className="text-[10px]">Trailing {trade.trailingStop} pips</TooltipContent>
   321→                                    </Tooltip>
   322→                                  ) : (
   323→                                    <Switch
   324→                                      checked={trade.isTrailingStop}
   325→                                      onCheckedChange={(checked) => handleToggleTrailing(trade.id, checked)}
   326→                                      className="scale-75 data-[state=checked]:bg-emerald-600"
   327→                                    />
   328→                                  )}
   329→                                </TableCell>
   330→                                <TableCell className="text-center">
   331→                                  <AnimatePresence>
   332→                                    {confirmClose === trade.id ? (
   333→                                      <motion.div
   334→                                        initial={{ opacity: 0, scale: 0.8 }}
   335→                                        animate={{ opacity: 1, scale: 1 }}
   336→                                        exit={{ opacity: 0, scale: 0.8 }}
   337→                                        className="flex items-center gap-1"
   338→                                      >
   339→                                        <Button
   340→                                          size="sm"
   341→                                          variant="destructive"
   342→                                          className="h-6 px-2 text-[10px]"
   343→                                          onClick={() => { closeTrade(trade.id); setConfirmClose(null); }}
   344→                                        >
   345→                                          Confirm
   346→                                        </Button>
   347→                                        <Button
   348→                                          size="sm"
   349→                                          variant="ghost"
   350→                                          className="h-6 px-2 text-[10px]"
   351→                                          onClick={() => setConfirmClose(null)}
   352→                                        >
   353→                                          <X className="h-3 w-3" />
   354→                                        </Button>

   356→                                    ) : (

   358→                                        size="sm"
   359→                                        variant="ghost"
   360→                                        className="h-6 px-2 text-muted-foreground hover:text-red-500 hover:bg-red-500/15 transition-colors duration-200"
   361→                                        onClick={() => setConfirmClose(trade.id)}
   362→                                      >
   363→                                        <Square className="h-3 w-3" />
   364→                                      </Button>
   365→                                    )}
   366→                                  </AnimatePresence>
   367→                                </TableCell>
   368→                              </TableRow>
   369→                              );
   370→                            })}
   371→                          </TableBody>
   372→                        </Table>
   373→                      </div>
   374→                    )}
   375→                  </CardContent>
   376→                </TabsContent>
   377→
   378→                <TabsContent value="history" className="mt-0">
   379→                  <CardContent className="px-4 pb-3">
   380→                    {closedTrades.length === 0 ? (
   381→                      <div className="text-center py-10 text-muted-foreground">
   382→                        <History className="h-8 w-8 mx-auto mb-2 opacity-30" />

   384→                        <p className="text-[11px] mt-1 opacity-60">Closed trades will appear here</p>

   386→                    ) : (
   387→                      <>
   388→                        {/* Summary Stats + Win/Loss Ratio Bar + Avg Comparison */}
   389→                        {historyStats && (() => {

   391→                          const losses = closedTrades.filter(t => t.profit <= 0);
   392→                          const avgWin = wins.length > 0 ? wins.reduce((s, t) => s + t.profit, 0) / wins.length : 0;

   394→                          const maxAvg = Math.max(avgWin, avgLoss, 1);
   395→                          return (
   396→                            <>
   397→                            <div className="grid grid-cols-5 gap-2 mb-3">
   398→                              <div className="text-center p-2 rounded-lg bg-slate-800/30 stat-card-pattern">

   404→                                <div className={`text-sm font-bold tabular-nums ${historyStats.winRate >= 50 ? 'text-emerald-500' : 'text-red-500'}`}>

   407→                              </div>
   408→                              <div className="text-center p-2 rounded-lg bg-slate-800/30 stat-card-pattern">

   428→                            <div className="mb-3">
