    18→function generateMockBacktestResult(strategy: StrategyName, symbol: Symbol): BacktestResult {
    19→  const initialBalance = 10000;
    20→  const totalTrades = Math.floor(Math.random() * 150) + 50;
    21→  const winRate = 40 + Math.random() * 30;
    22→  const wins = Math.floor(totalTrades * winRate / 100);
    23→  const losses = totalTrades - wins;
    24→  const avgWin = 15 + Math.random() * 30;
    25→  const avgLoss = 8 + Math.random() * 15;
    26→  const totalProfit = wins * avgWin;
    27→  const totalLoss = losses * avgLoss;
    28→  const finalBalance = initialBalance + totalProfit - totalLoss;
    29→  const maxDD = 5 + Math.random() * 20;
    30→  const sharpeRatio = 0.5 + Math.random() * 2.5;
    31→  const profitFactor = totalLoss > 0 ? totalProfit / totalLoss : 0;
    32→
    33→  const equityCurve: { trade: number; equity: number }[] = [];
    34→  let equity = initialBalance;
    35→  for (let i = 0; i < totalTrades; i++) {
    36→    const isWin = Math.random() * 100 < winRate;
    37→    equity += isWin ? avgWin * (0.5 + Math.random()) : -avgLoss * (0.5 + Math.random());
    38→    equityCurve.push({ trade: i + 1, equity: Math.round(equity * 100) / 100 });
    39→  }
    40→
    41→  return {
    42→    id: `bt-${Date.now()}`,
    43→    name: STRATEGIES[strategy].label,
    44→    symbol,
    45→    strategy,
    46→    initialBalance,
    47→    finalBalance: Math.round(finalBalance * 100) / 100,
    48→    totalTrades,
    49→    winRate: Math.round(winRate * 10) / 10,
    50→    profitFactor: Math.round(profitFactor * 100) / 100,
    51→    maxDrawdown: Math.round(maxDD * 10) / 10,
    52→    sharpeRatio: Math.round(sharpeRatio * 100) / 100,
    53→    totalProfit: Math.round(totalProfit * 100) / 100,
    54→    totalLoss: Math.round(totalLoss * 100) / 100,
    55→    avgWin: Math.round(avgWin * 100) / 100,
    56→    avgLoss: Math.round(avgLoss * 100) / 100,
    57→    equityCurve,
    58→  };
    59→}

    61→function WinRateRing({ value }: { value: number }) {
    62→  const radius = 20;
    63→  const circumference = 2 * Math.PI * radius;
    64→  const offset = circumference * (1 - value / 100);
    65→  const color = value >= 55 ? '#10b981' : value >= 45 ? '#f59e0b' : '#ef4444';
    66→
    67→  return (
    68→    <div className="relative w-12 h-12 flex-shrink-0">
    69→      <svg width="48" height="48" className="-rotate-90">
    70→        <circle cx="24" cy="24" r={radius} fill="none" stroke="rgba(255,255,255,0.06)" strokeWidth="3.5" />
    71→        <circle
    72→          cx="24" cy="24" r={radius} fill="none" stroke={color} strokeWidth="3.5"
    73→          strokeDasharray={circumference} strokeDashoffset={offset} strokeLinecap="round"
    74→          style={{ transition: 'stroke-dashoffset 0.6s ease' }}
    75→        />
    76→      </svg>
    77→      <div className="absolute inset-0 flex items-center justify-center">
    78→        <span className="text-[10px] font-bold tabular-nums" style={{ color }}>{Math.round(value)}</span>
    79→      </div>
    80→    </div>
    81→  );
    82→}
