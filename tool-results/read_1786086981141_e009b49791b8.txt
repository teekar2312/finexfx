    60→function generateMockBacktestResult(strategy: StrategyName, symbol: Symbol): BacktestResult {
    61→  const initialBalance = 10000;
    62→  const totalTrades = Math.floor(Math.random() * 150) + 50;
    63→  const winRate = 40 + Math.random() * 30;
    64→  const wins = Math.floor(totalTrades * winRate / 100);
    65→  const losses = totalTrades - wins;
    66→  const avgWin = 15 + Math.random() * 30;
    67→  const avgLoss = 8 + Math.random() * 15;
    68→  const totalProfit = wins * avgWin;
    69→  const totalLoss = losses * avgLoss;
    70→  const finalBalance = initialBalance + totalProfit - totalLoss;
    71→  const maxDD = 5 + Math.random() * 20;
    72→  const sharpeRatio = 0.5 + Math.random() * 2.5;
    73→  const profitFactor = totalLoss > 0 ? totalProfit / totalLoss : 0;
    74→
    75→  const equityCurve: { trade: number; equity: number }[] = [];
    76→  let equity = initialBalance;
    77→  for (let i = 0; i < totalTrades; i++) {
    78→    const isWin = Math.random() * 100 < winRate;
    79→    equity += isWin ? avgWin * (0.5 + Math.random()) : -avgLoss * (0.5 + Math.random());
    80→    equityCurve.push({ trade: i + 1, equity: Math.round(equity * 100) / 100 });
    81→  }
    82→
    83→  return {
    84→    id: `bt-${Date.now()}`,
    85→    name: STRATEGIES[strategy].label,
    86→    symbol,
    87→    strategy,
    88→    initialBalance,
    89→    finalBalance: Math.round(finalBalance * 100) / 100,
    90→    totalTrades,
    91→    winRate: Math.round(winRate * 10) / 10,
    92→    profitFactor: Math.round(profitFactor * 100) / 100,
    93→    maxDrawdown: Math.round(maxDD * 10) / 10,
    94→    sharpeRatio: Math.round(sharpeRatio * 100) / 100,
    95→    totalProfit: Math.round(totalProfit * 100) / 100,
    96→    totalLoss: Math.round(totalLoss * 100) / 100,
    97→    avgWin: Math.round(avgWin * 100) / 100,
    98→    avgLoss: Math.round(avgLoss * 100) / 100,
    99→    equityCurve,
   100→  };
   101→}

   103→function WinRateRing({ value }: { value: number }) {
   104→  const radius = 20;
   105→  const circumference = 2 * Math.PI * radius;
   106→  const offset = circumference * (1 - value / 100);
   107→  const color = value >= 55 ? '#10b981' : value >= 45 ? '#f59e0b' : '#ef4444';
   108→
   109→  return (
   110→    <div className="relative w-12 h-12 flex-shrink-0">
   111→      <svg width="48" height="48" className="-rotate-90">
   112→        <circle cx="24" cy="24" r={radius} fill="none" stroke="rgba(255,255,255,0.06)" strokeWidth="3.5" />
   113→        <circle
   114→          cx="24" cy="24" r={radius} fill="none" stroke={color} strokeWidth="3.5"
   115→          strokeDasharray={circumference} strokeDashoffset={offset} strokeLinecap="round"
   116→          style={{ transition: 'stroke-dashoffset 0.6s ease' }}
   117→        />
   118→      </svg>
   119→      <div className="absolute inset-0 flex items-center justify-center">
   120→        <span className="text-[10px] font-bold tabular-nums" style={{ color }}>{Math.round(value)}</span>
   121→      </div>
   122→    </div>
   123→  );
   124→}
