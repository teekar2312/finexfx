   665→                    disabled={logCounts.resolved === 0}
   666→                  >
   667→                    Clear Resolved
   668→                  </Button>
   669→                </div>
   670→              </div>
   671→
   672→              {/* Filter Buttons */}
   673→              <div className="flex items-center gap-2 mb-3">
   674→                {filterButtons.map((fb) => (
   675→                  <Button
   676→                    key={fb.key}
   677→                    variant={logFilter === fb.key ? 'default' : 'outline'}
   678→                    size="sm"
   679→                    className={`h-7 text-[10px] ${logFilter === fb.key ? 'bg-primary/15 text-primary border-primary/30 hover:bg-primary/20' : ''}`}

   686→                        {fb.count}
   687→                      </Badge>
   688→                    )}
   689→                  </Button>
   690→                ))}
   691→              </div>
   692→
   693→              {/* Log Entries - Timeline Format */}
   694→              {filteredLogs.length === 0 ? (
   695→                <div className="text-center py-8">
   696→                  <div className="w-14 h-14 rounded-full bg-accent/50 border border-border flex items-center justify-center mx-auto mb-3">

   701→                </div>

   703→                <ScrollArea className="h-[500px]">

   705→                    {/* Vertical timeline line */}
   706→                    <div className="absolute left-[7px] top-0 bottom-0 w-px bg-border" />
   707→
   708→                    <div className="space-y-2">
   709→                      {filteredLogs.map((log, i) => {
   710→                        const lineColor = log.level === 'error' ? 'bg-red-500' : log.level === 'warning' ? 'bg-amber-500' : 'bg-slate-500';
   711→                        const borderColor = log.level === 'error' ? 'border-red-500/20 bg-red-500/5' : log.level === 'warning' ? 'border-amber-500/20 bg-amber-500/5' : 'border-border bg-accent/30';
   712→                        return (

   714→                            key={log.id}
   715→                            initial={{ opacity: 0, x: -10 }}
   716→                            animate={{ opacity: 1, x: 0 }}
   717→                            transition={{ delay: i * 0.02 }}
   718→                            className="relative"
   719→                          >
   720→                            {/* Timeline dot */}
   721→                            <div className={`absolute -left-5 top-3 w-3.5 h-3.5 rounded-full ${lineColor} border-2 border-[#0a0f1c]`} />
   722→
   723→                            <div className={`p-3 rounded-lg border ${borderColor} ${log.resolved ? 'opacity-40' : ''}`}>
   724→                              <div className="flex items-center justify-between mb-1">
   725→                                <div className="flex items-center gap-2">
   726→                                  <Badge
   727→                                    variant="outline"
   728→                                    className={`text-[9px] px-1.5 py-0 ${
   729→                                      log.level === 'error' ? 'border-red-500/50 text-red-500' :
   730→                                      log.level === 'warning' ? 'border-amber-500/50 text-amber-500' :
   731→                                      'border-slate-500/50 text-slate-500'

   734→                                    {log.level.toUpperCase()}
   735→                                  </Badge>
   736→                                  <span className="text-[10px] text-muted-foreground">{log.source}</span>
   737→                                  {log.resolved && (
   738→                                    <Badge variant="outline" className="text-[9px] px-1 py-0 border-emerald-500/50 text-emerald-500">
   739→                                      RESOLVED
   740→                                    </Badge>
   741→                                  )}
   742→                                </div>
   743→                                <span className="text-[10px] text-muted-foreground tabular-nums">{log.timestamp}</span>
   744→                              </div>
   745→                              <div className="text-[11px] leading-relaxed">{log.message}</div>
   746→                            </div>
   747→                          </motion.div>
   748→                        );
   749→                      })}
   750→                    </div>
   751→                  </div>
   752→                </ScrollArea>
   753→              )}
   754→            </CardContent>
   755→          </Card>
   756→        </TabsContent>
   757→      </Tabs>
   758→    </div>
