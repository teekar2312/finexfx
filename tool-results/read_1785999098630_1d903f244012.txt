    24→    24→function getSessionStatus(session: { start: number; end: number }) {
    25→    25→  const now = new Date();
    26→    26→  const currentHour = now.getUTCHours();
    27→    27→  const currentMinute = now.getUTCMinutes();
    28→    28→  const currentDecimal = currentHour + currentMinute / 60;
    29→    29→  const isActive = currentDecimal >= session.start && currentDecimal < session.end;
    30→    30→  const progress = isActive
    31→    31→    ? ((currentDecimal - session.start) / (session.end - session.start)) * 100
    32→    32→    : 0;
    33→    33→  const duration = session.end - session.start;
    34→    34→  let statusText = '';
    35→    35→  if (isActive) {
    36→    36→    statusText = 'ACTIVE';
    37→    37→  } else if (currentDecimal < session.start) {
    38→    38→    const hoursUntil = session.start - currentHour;
    39→    39→    const minsUntil = (hoursUntil * 60 - currentMinute);
    40→    40→    if (minsUntil > 0) {
    41→    41→      const h = Math.floor(minsUntil / 60);
    42→    42→      const m = minsUntil % 60;
    43→    43→      statusText = `Opens in ${h}:${m.toString().padStart(2, '0')}`;
    44→    44→    } else {
    45→    45→      statusText = 'Closed';
    46→    46→    }
    47→    47→  } else {
    48→    48→    statusText = 'Closed';
    49→    49→  }
    50→    50→  return { isActive, progress, statusText, duration };
    51→    51→}

    53→    53→function timeAgo(dateStr: string): string {
    54→    54→  const now = Date.now();
    55→    55→  const then = new Date(dateStr).getTime();
    56→    56→  const diffMs = now - then;
    57→    57→  const diffSec = Math.floor(diffMs / 1000);
    58→    58→  if (diffSec < 60) return `${diffSec}s ago`;
    59→    59→  const diffMin = Math.floor(diffSec / 60);
    60→    60→  if (diffMin < 60) return `${diffMin}m ago`;
    61→    61→  const diffHr = Math.floor(diffMin / 60);
    62→    62→  return `${diffHr}h ago`;
    63→    63→}

    65→    65→function MiniSparkline({ values, width = 48, height = 18 }: { values: number[]; width?: number; height?: number }) {
    66→    66→  if (values.length < 2) return null;
    67→    67→  const min = Math.min(...values);
    68→    68→  const max = Math.max(...values);
    69→    69→  const range = max - min || 1;
    70→    70→  const padding = 2;
    71→    71→  const points = values.map((v, i) => {
    72→    72→    const x = padding + (i / (values.length - 1)) * (width - padding * 2);
    73→    73→    const y = padding + (1 - (v - min) / range) * (height - padding * 2);
    74→    74→    return `${x},${y}`;
    75→    75→  }).join(' ');
    76→    76→  const isUp = values[values.length - 1] >= values[0];
    77→    77→  const color = isUp ? '#10b981' : '#ef4444';
    78→    78→  return (
    79→    79→    <svg width={width} height={height} className="inline-block">
    80→    80→      <polyline
    81→    81→        fill="none"
    82→    82→        stroke={color}
    83→    83→        strokeWidth={1.5}
    84→    84→        strokeLinecap="round"
    85→    85→        strokeLinejoin="round"
    86→    86→        points={points}
    87→    87→      />
    88→    88→    </svg>
    89→    89→  );
    90→    90→}
