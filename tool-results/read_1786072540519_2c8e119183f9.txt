   175→  'Mean Revert',
   176→  'Trend Follow',
   177→];
   178→const DAY_LABELS = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri'];
   179→
   180→function generateWeeklyData(): WeekSummary[] {
   181→  const weeks: WeekSummary[] = [];
   182→  const baseDate = new Date(2025, 0, 6); // Monday Jan 6 2025
   183→
   184→  for (let w = 0; w < 4; w++) {
   185→    const rng = createSeededRandom(42 + w * 137);
   186→    const days: DayData[] = [];
   187→    let weekPnl = 0;
   188→    let weekTrades = 0;
   189→    let weekWins = 0;
   190→
   191→    for (let d = 0; d < 5; d++) {
   192→      const date = new Date(baseDate);
   193→      date.setDate(date.getDate() + w * 7 + d);
   194→      const dateStr = date.toLocaleDateString('en-US', {
   195→        month: 'short',
   196→        day: 'numeric',
   197→      });
   198→
   199→      const trades = Math.floor(rng() * 8) + 3; // 3-10 trades
   200→      const winRate = 35 + rng() * 40; // 35-75%
   201→      const pnlBase = (rng() - 0.38) * 120; // slight positive bias
   202→      const pnl = Math.round(pnlBase * 10) / 10;
   203→      const bestPair = PAIRS[Math.floor(rng() * PAIRS.length)];
   204→      const bestStrategy = STRATEGIES[Math.floor(rng() * STRATEGIES.length)];
   205→
   206→      // Generate 5 intraday points for sparkline
   207→      const sparkline: number[] = [];
   208→      let running = 0;
   209→      for (let s = 0; s < 5; s++) {
   210→        running += (rng() - 0.4) * 30;
   211→        sparkline.push(Math.round(running * 10) / 10);
   212→      }
   213→
   214→      weekPnl += pnl;
   215→      weekTrades += trades;
   216→      weekWins += trades * (winRate / 100);
   217→
   218→      days.push({
   219→        date: dateStr,
   220→        dayLabel: DAY_LABELS[d],
   221→        trades,
   222→        winRate,
   223→        pnl,
   224→        bestPair,
   225→        bestStrategy,
   226→        sparkline,
   227→      });
   228→    }
   229→
   230→    const weekWinRate = Math.round((weekWins / weekTrades) * 1000) / 10;
   231→    const avgDailyPnl = Math.round((weekPnl / 5) * 10) / 10;
   232→
   233→    // Sharpe-like score: reward high avg pnl, penalize variance
   234→    const mean = weekPnl / 5;
   235→    const variance =
   236→      days.reduce((acc, day) => acc + Math.pow(day.pnl - mean, 2), 0) / 5;
   237→    const std = Math.sqrt(variance) || 1;
   238→    const sharpeScore = Math.round((mean / std) * 100) / 100;
   239→
   240→    const sorted = [...days].sort((a, b) => b.pnl - a.pnl);
   241→    const bestDay = `${sorted[0].dayLabel} (+$${sorted[0].pnl.toFixed(0)})`;
   242→    const worstDay = `${sorted[sorted.length - 1].dayLabel} ($${sorted[sorted.length - 1].pnl.toFixed(0)})`;
   243→
   244→    const sparkline = days.map((d) => d.pnl);
   245→
   246→    weeks.push({
   247→      weekIndex: w,
   248→      label: `Week ${w + 1}`,
   249→      totalPnl: Math.round(weekPnl * 10) / 10,
   250→      avgDailyPnl,
   251→      bestDay,
   252→      worstDay,
   253→      winRate: weekWinRate,
   254→      sharpeScore,
   255→      sparkline,
   256→      days,
   257→    });
   258→  }
   259→
   260→  return weeks;
   261→}
   262→
   263→function computeMonthlySummary(weeks: WeekSummary[]): MonthlySummary {
   264→  const totalPnl = Math.round(
   265→    weeks.reduce((s, w) => s + w.totalPnl, 0) * 10
   266→  ) / 10;
   267→  const totalTrades = weeks.reduce(
   268→    (s, w) => s + w.days.reduce((ds, d) => ds + d.trades, 0),
   269→    0
   270→  );
   271→  const totalWins = weeks.reduce(
   272→    (s, w) => s + w.days.reduce((ds, d) => ds + d.trades * (d.winRate / 100), 0),
   273→    0
   274→  );
   275→  const winRate = Math.round((totalWins / totalTrades) * 1000) / 10;
   276→
   277→  const bestWeekIdx = weeks.reduce(
   278→    (best, w, i) => (w.totalPnl > weeks[best].totalPnl ? i : best),
   279→    0
   280→  );
   281→
   282→  // Consistency: how many weeks are profitable
   283→  const profitableWeeks = weeks.filter((w) => w.totalPnl > 0).length;
   284→  const consistencyScore = Math.round((profitableWeeks / 4) * 100);
   285→
   286→  // Grade calculation
   287→  let grade: string;
   288→  let gradeColor: 'emerald' | 'amber' | 'red';
   289→
   290→  if (totalPnl > 500 && consistencyScore > 65) {
   291→    grade = 'A+';
   292→    gradeColor = 'emerald';
   293→  } else if (totalPnl > 350 && consistencyScore >= 50) {
   294→    grade = 'A';
   295→    gradeColor = 'emerald';
   296→  } else if (totalPnl > 200 && consistencyScore >= 50) {
   297→    grade = 'B+';
   298→    gradeColor = 'emerald';
   299→  } else if (totalPnl > 100) {
   300→    grade = 'B';
   301→    gradeColor = 'amber';
   302→  } else if (totalPnl > 0 && consistencyScore >= 50) {
   303→    grade = 'C';
   304→    gradeColor = 'amber';
   305→  } else if (totalPnl > -50) {
   306→    grade = 'C-';
   307→    gradeColor = 'amber';
   308→  } else if (totalPnl > -200) {
   309→    grade = 'D';
   310→    gradeColor = 'red';
   311→  } else {
   312→    grade = 'F';
   313→    gradeColor = 'red';
   314→  }
   315→
   316→  return {
   317→    totalPnl,
   318→    totalTrades,
   319→    winRate,
   320→    bestWeek: bestWeekIdx,
   321→    consistencyScore,
   322→    grade,
   323→    gradeColor,
   324→  };
   325→}

   327→// ─── Helpers ──────────────────────────────────────────────────────────────────
   328→
   329→function pnlColor(value: number): string {
   330→  if (value > 0) return 'text-emerald-400';
   331→  if (value < 0) return 'text-red-400';
   332→  return 'text-amber-400';
   333→}
   334→
   335→function pnlBg(value: number): string {
   336→  if (value > 0) return 'bg-emerald-500/10';
   337→  if (value < 0) return 'bg-red-500/10';
   338→  return 'bg-amber-500/10';
   339→}
   340→
   341→function pnlSign(value: number): string {
   342→  return value > 0 ? '+' : '';
   343→}
   344→
   345→function formatPnl(value: number): string {
   346→  return `${pnlSign(value)}$${value.toFixed(1)}`;
   347→}
