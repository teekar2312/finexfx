    56→    56→function timeAgo(dateStr: string): string {
    57→    57→  const now = Date.now();
    58→    58→  const then = new Date(dateStr).getTime();
    59→    59→  const diffMs = now - then;
    60→    60→  const diffSec = Math.floor(diffMs / 1000);
    61→    61→  if (diffSec < 60) return `${diffSec}s ago`;
    62→    62→  const diffMin = Math.floor(diffSec / 60);
    63→    63→  if (diffMin < 60) return `${diffMin}m ago`;
    64→    64→  const diffHr = Math.floor(diffMin / 60);
    65→    65→  return `${diffHr}h ago`;
    66→    66→}

    68→    68→function MiniSparkline({ values, width = 48, height = 18 }: { values: number[]; width?: number; height?: number }) {
    69→    69→  if (values.length < 2) return null;
    70→    70→  const min = Math.min(...values);
    71→    71→  const max = Math.max(...values);
    72→    72→  const range = max - min || 1;
    73→    73→  const padding = 2;
    74→    74→  const points = values.map((v, i) => {
    75→    75→    const x = padding + (i / (values.length - 1)) * (width - padding * 2);
    76→    76→    const y = padding + (1 - (v - min) / range) * (height - padding * 2);
    77→    77→    return `${x},${y}`;
    78→    78→  }).join(' ');
    79→    79→  const isUp = values[values.length - 1] >= values[0];
    80→    80→  const color = isUp ? '#10b981' : '#ef4444';
    81→    81→  return (
    82→    82→    <svg width={width} height={height} className="inline-block">
    83→    83→      <polyline
    84→    84→        fill="none"
    85→    85→        stroke={color}
    86→    86→        strokeWidth={1.5}
    87→    87→        strokeLinecap="round"
    88→    88→        strokeLinejoin="round"
    89→    89→        points={points}
    90→    90→      />
    91→    91→    </svg>
    92→    92→  );
    93→    93→}
