    50→// --- Animated Ring Gauge ---
    51→function ConfidenceRing({ confidence, size = 72 }: { confidence: number; size?: number }) {
    52→  const strokeWidth = 5;
    53→  const radius = (size - strokeWidth) / 2;
    54→  const circumference = 2 * Math.PI * radius;
    55→  const progress = (confidence / 100) * circumference;
    56→  const color = confidence >= 80 ? '#10b981' : confidence >= 60 ? '#f59e0b' : '#ef4444';
    57→  const center = size / 2;
    58→
    59→  return (
    60→    <div className="relative" style={{ width: size, height: size }}>
    61→      <svg width={size} height={size} className="-rotate-90">
    62→        <circle
    63→          cx={center}
    64→          cy={center}
    65→          r={radius}
    66→          fill="none"
    67→          stroke="currentColor"
    68→          strokeWidth={strokeWidth}
    69→          className="text-muted/30"
    70→        />
    71→        <motion.circle
    72→          cx={center}
    73→          cy={center}
    74→          r={radius}
    75→          fill="none"
    76→          stroke={color}
    77→          strokeWidth={strokeWidth}
    78→          strokeLinecap="round"
    79→          strokeDasharray={circumference}
    80→          initial={{ strokeDashoffset: circumference }}
    81→          animate={{ strokeDashoffset: circumference - progress }}
    82→          transition={{ duration: 0.8, ease: 'easeOut' }}
    83→        />
    84→      </svg>
    85→      <div className="absolute inset-0 flex items-center justify-center">
    86→        <span className="text-lg font-bold tabular-nums" style={{ color }}>
    87→          {confidence}%
    88→        </span>
    89→      </div>
    90→    </div>
    91→  );
    92→}

    94→// --- Strategy signal generation (deterministic for consistency) ---
    95→function getStrategySignalForSymbol(
    96→  strategy: StrategyName,
    97→  symbol: string,
    98→  mainDirection: TradeDirection | 'HOLD'
    99→): 'bullish' | 'bearish' | 'neutral' {
   100→  const seed = (strategy.length * 7 + symbol.length * 13) % 100;
   101→  if (seed < 55) return mainDirection === 'BUY' ? 'bullish' : mainDirection === 'SELL' ? 'bearish' : 'neutral';
   102→  if (seed < 75) return mainDirection === 'BUY' ? 'bearish' : mainDirection === 'SELL' ? 'bullish' : 'neutral';
   103→  return 'neutral';
   104→}
   105→
   106→function getStrategyConfidence(strategy: StrategyName, symbol: string): number {
   107→  const seed = (strategy.length * 11 + symbol.length * 17) % 100;
   108→  return 40 + seed * 0.5;
   109→}

   111→// --- Indicator data generation (deterministic) ---
   112→function generateIndicatorsForSignal(signal: TradingSignal) {
   113→  const allIndicators = [
   114→    { name: 'RSI(14)', value: 62.4, signal: 'bullish' as const },
   115→    { name: 'MACD(12,26,9)', value: 0.0032, signal: 'bullish' as const },
   116→    { name: 'EMA(9)', value: 1.0845, signal: 'bullish' as const },
   117→    { name: 'EMA(21)', value: 1.0838, signal: 'bullish' as const },
   118→    { name: 'ATR(14)', value: 0.0045, signal: 'neutral' as const },
   119→    { name: 'Bollinger Bands', value: 1.0850, signal: 'neutral' as const },
   120→    { name: 'Stochastic(14,3)', value: 71.2, signal: 'bullish' as const },
   121→    { name: 'SuperTrend(10,3)', value: 1.0830, signal: 'bearish' as const },
   122→    { name: 'Volume', value: 1523, signal: 'bullish' as const },
   123→    { name: 'CCI(20)', value: 85.3, signal: 'bullish' as const },
   124→  ];
   125→
   126→  const seed = (signal.id.charCodeAt(0) * 3 + signal.symbol.length * 7) % allIndicators.length;
   127→  const rotated = [...allIndicators.slice(seed), ...allIndicators.slice(0, seed)];
   128→
   129→  return rotated.slice(0, 8).map((ind) => {
   130→    const isBuy = signal.direction === 'BUY';
   131→    const s = (ind.name.length * 5 + signal.symbol.length * 11) % 10;
   132→    let sig = ind.signal;
   133→    if (s < 2) sig = isBuy ? 'bearish' : 'bullish';
   134→    else if (s < 4) sig = 'neutral';
   135→    return { ...ind, signal: sig };
   136→  });
   137→}
   138→
   139→// --- Historical accuracy simulation ---
   140→function generateHistoricalAccuracy(signal: TradingSignal) {
   141→  const seed = (signal.symbol.length * 13 + signal.strategy.length * 7) % 100;
   142→  const winRate = 55 + (seed % 25);
   143→  const lastSignals = Array.from({ length: 10 }, (_, i) => {
   144→    const s = (seed + i * 17) % 10;
   145→    return s < winRate / 10 ? 'win' : 'loss';
   146→  });
   147→  const avgPnl = (winRate - 50) * 1.8 + (seed % 5) - 2;
   148→  const bestTrade = 20 + seed % 60;
   149→  const worstTrade = -(10 + seed % 25);
   150→  return { winRate, lastSignals, avgPnl, bestTrade, worstTrade };
   151→}
