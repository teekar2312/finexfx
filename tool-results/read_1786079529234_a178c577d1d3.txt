    29→function getSessionStatus(session: { start: number; end: number }) {
    30→  const now = new Date();
    31→  const currentHour = now.getUTCHours();
    32→  const currentMinute = now.getUTCMinutes();
    33→  const currentDecimal = currentHour + currentMinute / 60;
    34→  const isActive = currentDecimal >= session.start && currentDecimal < session.end;
    35→  const progress = isActive
    36→    ? ((currentDecimal - session.start) / (session.end - session.start)) * 100
    37→    : 0;
    38→  const duration = session.end - session.start;
    39→  let statusText = '';
    40→  if (isActive) {
    41→    statusText = 'ACTIVE';
    42→  } else if (currentDecimal < session.start) {
    43→    const hoursUntil = session.start - currentHour;
    44→    const minsUntil = (hoursUntil * 60 - currentMinute);
    45→    if (minsUntil > 0) {
    46→      const h = Math.floor(minsUntil / 60);
    47→      const m = minsUntil % 60;
    48→      statusText = `Opens in ${h}:${m.toString().padStart(2, '0')}`;
    49→    } else {
    50→      statusText = 'Closed';
    51→    }
    52→  } else {
    53→    statusText = 'Closed';
    54→  }
    55→  return { isActive, progress, statusText, duration };
    56→}

    58→function timeAgo(dateStr: string): string {
    59→  const now = Date.now();
    60→  const then = new Date(dateStr).getTime();
    61→  const diffMs = now - then;
    62→  const diffSec = Math.floor(diffMs / 1000);
    63→  if (diffSec < 60) return `${diffSec}s ago`;
    64→  const diffMin = Math.floor(diffSec / 60);
    65→  if (diffMin < 60) return `${diffMin}m ago`;
    66→  const diffHr = Math.floor(diffMin / 60);
    67→  return `${diffHr}h ago`;
    68→}

    70→function MiniSparkline({ values, width = 48, height = 18 }: { values: number[]; width?: number; height?: number }) {
    71→  if (values.length < 2) return null;
    72→  const min = Math.min(...values);
    73→  const max = Math.max(...values);
    74→  const range = max - min || 1;
    75→  const padding = 2;
    76→  const points = values.map((v, i) => {
    77→    const x = padding + (i / (values.length - 1)) * (width - padding * 2);
    78→    const y = padding + (1 - (v - min) / range) * (height - padding * 2);
    79→    return `${x},${y}`;
    80→  }).join(' ');
    81→  const isUp = values[values.length - 1] >= values[0];
    82→  const color = isUp ? '#10b981' : '#ef4444';
    83→  return (
    84→    <svg width={width} height={height} className="inline-block">
    85→      <polyline
    86→        fill="none"
    87→        stroke={color}
    88→        strokeWidth={1.5}
    89→        strokeLinecap="round"
    90→        strokeLinejoin="round"
    91→        points={points}
    92→      />
    93→    </svg>
    94→  );
    95→}
