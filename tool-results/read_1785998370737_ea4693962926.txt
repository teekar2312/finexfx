    24→function getSessionStatus(session: { start: number; end: number }) {
    25→  const now = new Date();
    26→  const currentHour = now.getUTCHours();
    27→  const currentMinute = now.getUTCMinutes();
    28→  const currentDecimal = currentHour + currentMinute / 60;
    29→  const isActive = currentDecimal >= session.start && currentDecimal < session.end;
    30→  const progress = isActive
    31→    ? ((currentDecimal - session.start) / (session.end - session.start)) * 100
    32→    : 0;
    33→  const duration = session.end - session.start;
    34→  let statusText = '';
    35→  if (isActive) {
    36→    statusText = 'ACTIVE';
    37→  } else if (currentDecimal < session.start) {
    38→    const hoursUntil = session.start - currentHour;
    39→    const minsUntil = (hoursUntil * 60 - currentMinute);
    40→    if (minsUntil > 0) {
    41→      const h = Math.floor(minsUntil / 60);
    42→      const m = minsUntil % 60;
    43→      statusText = `Opens in ${h}:${m.toString().padStart(2, '0')}`;
    44→    } else {
    45→      statusText = 'Closed';
    46→    }
    47→  } else {
    48→    statusText = 'Closed';
    49→  }
    50→  return { isActive, progress, statusText, duration };
    51→}

    53→function timeAgo(dateStr: string): string {
    54→  const now = Date.now();
    55→  const then = new Date(dateStr).getTime();
    56→  const diffMs = now - then;
    57→  const diffSec = Math.floor(diffMs / 1000);
    58→  if (diffSec < 60) return `${diffSec}s ago`;
    59→  const diffMin = Math.floor(diffSec / 60);
    60→  if (diffMin < 60) return `${diffMin}m ago`;
    61→  const diffHr = Math.floor(diffMin / 60);
    62→  return `${diffHr}h ago`;
    63→}

    65→function MiniSparkline({ values, width = 48, height = 18 }: { values: number[]; width?: number; height?: number }) {
    66→  if (values.length < 2) return null;
    67→  const min = Math.min(...values);
    68→  const max = Math.max(...values);
    69→  const range = max - min || 1;
    70→  const padding = 2;
    71→  const points = values.map((v, i) => {
    72→    const x = padding + (i / (values.length - 1)) * (width - padding * 2);
    73→    const y = padding + (1 - (v - min) / range) * (height - padding * 2);
    74→    return `${x},${y}`;
    75→  }).join(' ');
    76→  const isUp = values[values.length - 1] >= values[0];
    77→  const color = isUp ? '#10b981' : '#ef4444';
    78→  return (
    79→    <svg width={width} height={height} className="inline-block">
    80→      <polyline
    81→        fill="none"
    82→        stroke={color}
    83→        strokeWidth={1.5}
    84→        strokeLinecap="round"
    85→        strokeLinejoin="round"
    86→        points={points}
    87→      />
    88→    </svg>
    89→  );
    90→}
