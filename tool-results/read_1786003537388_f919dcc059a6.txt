   564→function JournalAnalytics({ entries }: { entries: JournalEntry[] }) {
   565→  const wins = entries.filter((e) => e.pnl >= 0);
   566→  const losses = entries.filter((e) => e.pnl < 0);
   567→  const winRate = entries.length > 0 ? (wins.length / entries.length) * 100 : 0;
   568→  const totalPnl = entries.reduce((s, e) => s + e.pnl, 0);
   569→  const avgPnl = entries.length > 0 ? totalPnl / entries.length : 0;
   570→  const avgRating = entries.length > 0 ? entries.reduce((s, e) => s + e.rating, 0) / entries.length : 0;
   571→  const avgWin = wins.length > 0 ? wins.reduce((s, e) => s + e.pnl, 0) / wins.length : 0;
   572→  const avgLoss = losses.length > 0 ? losses.reduce((s, e) => s + e.pnl, 0) / losses.length : 0;
   573→
   574→  // Strategy performance
   575→  const strategyMap = new Map<string, { count: number; pnl: number; wins: number }>();
   576→  entries.forEach((e) => {
   577→    const s = strategyMap.get(e.strategy) || { count: 0, pnl: 0, wins: 0 };
   578→    s.count++;
   579→    s.pnl += e.pnl;
   580→    if (e.pnl >= 0) s.wins++;
   581→    strategyMap.set(e.strategy, s);
   582→  });
   583→  const strategyData = Array.from(strategyMap.entries()).map(([name, data]) => ({
   584→    name,
   585→    trades: data.count,
   586→    pnl: parseFloat(data.pnl.toFixed(2)),
   587→    winRate: parseFloat(((data.wins / data.count) * 100).toFixed(0)),
   588→  }));
   589→
   590→  // Mood distribution
   591→  const moodMap = new Map<string, number>();
   592→  entries.forEach((e) => moodMap.set(e.mood, (moodMap.get(e.mood) || 0) + 1));

   594→    name: MOOD_CONFIG[m].label,
   595→    value: moodMap.get(m) || 0,
   596→  })).filter((d) => d.value > 0);
   597→
   598→  // P&L by symbol

   602→    name: SYMBOL_DISPLAY[sym] || sym,

   619→            <div className="text-[10px] text-muted-foreground mt-0.5">{wins.length}W / {losses.length}L</div>

   802→  const [sortDir, setSortDir] = useState<'desc' | 'asc'>('desc');
   803→  const [showAnalytics, setShowAnalytics] = useState(false);
   804→  const [dialogOpen, setDialogOpen] = useState(false);
   805→  const [editingEntry, setEditingEntry] = useState<JournalEntry | undefined>(undefined);
   806→  const [deleteConfirm, setDeleteConfirm] = useState<string | null>(null);
   807→
   808→  // Extract unique strategies from entries for filter
   809→  const allStrategies = useMemo(() => {
   810→    const set = new Set(journalEntries.map((e) => e.strategy));
   811→    return Array.from(set).sort();
   812→  }, [journalEntries]);
   813→
   814→  // Filter and sort entries
   815→  const filteredEntries = useMemo(() => {
   816→    let result = [...journalEntries];
   817→
   818→    if (searchQuery) {
   819→      const q = searchQuery.toLowerCase();
   820→      result = result.filter(
   821→        (e) =>
   822→          e.symbol.toLowerCase().includes(q) ||
   823→          e.strategy.toLowerCase().includes(q) ||
   824→          e.notes.toLowerCase().includes(q) ||
   825→          e.tags.some((t) => t.toLowerCase().includes(q)) ||
   826→          e.lessons.toLowerCase().includes(q)
   827→      );
   828→    }
   829→
   830→    if (filterSymbol !== 'all') {
   831→      result = result.filter((e) => e.symbol === filterSymbol);
   832→    }
   833→    if (filterMood !== 'all') {
   834→      result = result.filter((e) => e.mood === filterMood);
   835→    }
   836→    if (filterStrategy !== 'all') {
   837→      result = result.filter((e) => e.strategy === filterStrategy);
   838→    }
   839→
   840→    result.sort((a, b) => {
   841→      const dir = sortDir === 'desc' ? -1 : 1;
   842→      switch (sortBy) {
   843→        case 'date': return dir * (new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime());
   844→        case 'pnl': return dir * (a.pnl - b.pnl);
   845→        case 'pips': return dir * (a.pips - b.pips);
   846→        case 'rating': return dir * (a.rating - b.rating);
   847→        default: return 0;
   848→      }
   849→    });
   850→
   851→    return result;
   852→  }, [journalEntries, searchQuery, filterSymbol, filterMood, filterStrategy, sortBy, sortDir]);
   853→
   854→  const handleAddEntry = (data: Omit<JournalEntry, 'id' | 'createdAt'>) => {
   855→    addJournalEntry({
   856→      ...data,
   857→      id: `j-${Date.now()}`,
   858→      createdAt: new Date().toISOString(),
   859→    });
   860→    setDialogOpen(false);
   861→    setEditingEntry(undefined);
   862→    addNotification({ type: 'success', title: 'Journal Entry Added', message: `${data.direction} ${data.symbol} logged in journal` });
   863→  };
   864→
   865→  const handleUpdateEntry = (data: Omit<JournalEntry, 'id' | 'createdAt'>) => {
   866→    if (editingEntry) {
   867→      updateJournalEntry(editingEntry.id, data);
   868→      setDialogOpen(false);
   869→      setEditingEntry(undefined);
   870→      addNotification({ type: 'success', title: 'Entry Updated', message: 'Journal entry has been updated' });
   871→    }
   872→  };
   873→
   874→  const handleEdit = (entry: JournalEntry) => {
   875→    setEditingEntry(entry);
   876→    setDialogOpen(true);
   877→  };
   878→
   879→  const handleDelete = (id: string) => {
   880→    deleteJournalEntry(id);
   881→    setDeleteConfirm(null);
   882→    addNotification({ type: 'info', title: 'Entry Deleted', message: 'Journal entry removed' });
   883→  };
   884→
   885→  const handleSave = editingEntry ? handleUpdateEntry : handleAddEntry;
   886→
   887→  const handleToggleSort = (field: typeof sortBy) => {
   888→    if (sortBy === field) {
   889→      setSortDir(sortDir === 'desc' ? 'asc' : 'desc');
   890→    } else {
   891→      setSortBy(field);
   892→      setSortDir('desc');
   893→    }
   894→  };
   895→
   896→  const hasActiveFilters = searchQuery || filterSymbol !== 'all' || filterMood !== 'all' || filterStrategy !== 'all';
   897→
   898→  return (
   899→    <div className="p-4 md:p-6 space-y-6">
   900→      {/* Header */}
   901→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   902→        <div>
   903→          <div className="flex items-center gap-2.5">
   904→            <div className="w-9 h-9 rounded-lg bg-primary/10 flex items-center justify-center">
   905→              <BookOpen className="h-5 w-5 text-primary" />
   906→            </div>
   907→            <div>
   908→              <h1 className="text-lg font-bold">Trade Journal</h1>
   909→              <p className="text-xs text-muted-foreground">Review, annotate, and learn from your closed trades</p>
   910→            </div>
   911→          </div>
   912→        </div>
   913→        <div className="flex items-center gap-2">
   914→          <Button
   915→            variant={showAnalytics ? 'default' : 'outline'}
   916→            size="sm"
   917→            className="h-8 text-xs gap-1.5"
   918→            onClick={() => setShowAnalytics(!showAnalytics)}
   919→          >
   920→            <BarChart3 className="h-3.5 w-3.5" />
   921→            Analytics
   922→          </Button>
   923→          <Dialog open={dialogOpen} onOpenChange={(open) => { setDialogOpen(open); if (!open) setEditingEntry(undefined); }}>
   924→            <DialogTrigger asChild>
   925→              <Button size="sm" className="h-8 text-xs gap-1.5">
   926→                <Plus className="h-3.5 w-3.5" />
   927→                New Entry
   928→              </Button>
   929→            </DialogTrigger>
   930→            <DialogContent className="max-w-lg max-h-[85vh] overflow-y-auto">
   931→              <DialogHeader>
   932→                <DialogTitle className="text-sm">{editingEntry ? 'Edit Journal Entry' : 'New Journal Entry'}</DialogTitle>
   933→              </DialogHeader>
   934→              <JournalEntryForm
   935→                entry={editingEntry}
   936→                onSave={handleSave}
   937→                onCancel={() => { setDialogOpen(false); setEditingEntry(undefined); }}
   938→              />
   939→            </DialogContent>
   940→          </Dialog>
   941→        </div>
   942→      </div>
   943→
   944→      {/* Analytics Panel */}
   945→      <AnimatePresence>
   946→        {showAnalytics && (
   947→          <motion.div
   948→            initial={{ height: 0, opacity: 0 }}
   949→            animate={{ height: 'auto', opacity: 1 }}
   950→            exit={{ height: 0, opacity: 0 }}
   951→            transition={{ duration: 0.3 }}
   952→            className="overflow-hidden"
   953→          >
   954→            <JournalAnalytics entries={journalEntries} />
   955→          </motion.div>
   956→        )}
   957→      </AnimatePresence>
   958→
   959→      {/* Filters Bar */}
   960→      <Card className="glass-card border-0 p-0">
   961→        <CardContent className="p-3">
   962→          <div className="flex flex-col md:flex-row gap-3">
   963→            {/* Search */}
   964→            <div className="relative flex-1 min-w-0">
   965→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   966→              <Input
   967→                value={searchQuery}
   968→                onChange={(e) => setSearchQuery(e.target.value)}
   969→                placeholder="Search notes, tags, strategies..."
   970→                className="h-8 pl-8 text-xs"
   971→              />
   972→              {searchQuery && (
   973→                <button
   974→                  onClick={() => setSearchQuery('')}
   975→                  className="absolute right-2 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   976→                >
   977→                  <X className="h-3 w-3" />
   978→                </button>
   979→              )}
   980→            </div>
   981→
   982→            {/* Symbol Filter */}
   983→            <Select value={filterSymbol} onValueChange={setFilterSymbol}>
   984→              <SelectTrigger className="h-8 w-full md:w-[130px] text-xs">
   985→                <SelectValue placeholder="Symbol" />
   986→              </SelectTrigger>
   987→              <SelectContent>
   988→                <SelectItem value="all" className="text-xs">All Symbols</SelectItem>
   989→                {SYMBOLS_LIST.map((s) => (
   990→                  <SelectItem key={s} value={s} className="text-xs">{SYMBOL_DISPLAY[s]}</SelectItem>
   991→                ))}
   992→              </SelectContent>
   993→            </Select>
   994→
   995→            {/* Mood Filter */}
   996→            <Select value={filterMood} onValueChange={setFilterMood}>
   997→              <SelectTrigger className="h-8 w-full md:w-[120px] text-xs">
   998→                <SelectValue placeholder="Mood" />
   999→              </SelectTrigger>
  1000→              <SelectContent>
  1001→                <SelectItem value="all" className="text-xs">All Moods</SelectItem>
  1002→                {(Object.keys(MOOD_CONFIG) as Array<JournalEntry['mood']>).map((m) => (
  1003→                  <SelectItem key={m} value={m} className="text-xs">
  1004→                    {MOOD_CONFIG[m].emoji} {MOOD_CONFIG[m].label}
  1005→                  </SelectItem>
  1006→                ))}
  1007→              </SelectContent>
  1008→            </Select>
  1009→
  1010→            {/* Strategy Filter */}
  1011→            <Select value={filterStrategy} onValueChange={setFilterStrategy}>
  1012→              <SelectTrigger className="h-8 w-full md:w-[150px] text-xs">
  1013→                <SelectValue placeholder="Strategy" />
  1014→              </SelectTrigger>
  1015→              <SelectContent>
  1016→                <SelectItem value="all" className="text-xs">All Strategies</SelectItem>
  1017→                {allStrategies.map((s) => (
  1018→                  <SelectItem key={s} value={s} className="text-xs">{s}</SelectItem>
  1019→                ))}
  1020→              </SelectContent>
  1021→            </Select>
  1022→          </div>
  1023→
  1024→          {/* Sort Options + Clear */}
  1025→          <div className="flex items-center justify-between mt-2.5 pt-2.5 border-t border-border/50">
  1026→            <div className="flex items-center gap-1">
  1027→              <Filter className="h-3 w-3 text-muted-foreground" />
  1028→              <span className="text-[10px] text-muted-foreground mr-1">Sort:</span>
  1029→              {([['date', 'Date'], ['pnl', 'P&L'], ['pips', 'Pips'], ['rating', 'Rating']] as const).map(([key, label]) => (
  1030→                <button
  1031→                  key={key}
  1032→                  onClick={() => handleToggleSort(key)}
  1033→                  className={`text-[10px] px-2 py-0.5 rounded-md transition-colors
  1034→                    ${sortBy === key
  1035→                      ? 'bg-primary/10 text-primary font-medium'
  1036→                      : 'text-muted-foreground hover:text-foreground hover:bg-accent'}`}
  1037→                >
  1038→                  {label}
  1039→                  {sortBy === key && (
  1040→                    <span className="ml-0.5">{sortDir === 'desc' ? '↓' : '↑'}</span>
  1041→                  )}
  1042→                </button>
  1043→              ))}
  1044→            </div>
  1045→            <div className="flex items-center gap-2">
  1046→              {hasActiveFilters && (
  1047→                <button
  1048→                  onClick={() => { setSearchQuery(''); setFilterSymbol('all'); setFilterMood('all'); setFilterStrategy('all'); }}
  1049→                  className="text-[10px] text-muted-foreground hover:text-foreground transition-colors"
  1050→                >
  1051→                  Clear filters
  1052→                </button>
  1053→              )}
  1054→              <span className="text-[10px] text-muted-foreground">
  1055→                {filteredEntries.length} {filteredEntries.length === 1 ? 'entry' : 'entries'}
  1056→              </span>
  1057→            </div>
  1058→          </div>
  1059→        </CardContent>
  1060→      </Card>
  1061→
  1062→      {/* Journal Entries List */}
  1063→      <div className="space-y-3">
  1064→        {filteredEntries.length === 0 ? (
  1065→          <div className="glass-card rounded-lg p-12 text-center">
  1066→            <BookOpen className="h-10 w-10 text-muted-foreground mx-auto mb-3" />
  1067→            <h3 className="text-sm font-medium text-muted-foreground mb-1">
  1068→              {hasActiveFilters ? 'No matching entries' : 'No journal entries yet'}
  1069→            </h3>
  1070→            <p className="text-xs text-muted-foreground">
  1071→              {hasActiveFilters
  1072→                ? 'Try adjusting your filters to find what you\'re looking for.'
  1073→                : 'Start documenting your trades to build a trading diary and improve your performance.'}
  1074→            </p>
  1075→            {!hasActiveFilters && (
  1076→              <Button
  1077→                size="sm"
  1078→                className="mt-4 h-8 text-xs gap-1.5"
  1079→                onClick={() => setDialogOpen(true)}
  1080→              >
  1081→                <Plus className="h-3.5 w-3.5" />
  1082→                Add First Entry
  1083→              </Button>
  1084→            )}
  1085→          </div>
  1086→        ) : (
  1087→          filteredEntries.map((entry, i) => (
  1088→            <JournalEntryCard
  1089→              key={entry.id}
  1090→              entry={entry}
  1091→              index={i}
  1092→              onEdit={handleEdit}
  1093→              onDelete={(id) => setDeleteConfirm(id)}
  1094→            />
  1095→          ))
  1096→        )}
  1097→      </div>
  1098→
  1099→      {/* Delete Confirmation Dialog */}
  1100→      <Dialog open={!!deleteConfirm} onOpenChange={(open) => !open && setDeleteConfirm(null)}>
  1101→        <DialogContent className="max-w-sm">
  1102→          <DialogHeader>
  1103→            <DialogTitle className="text-sm">Delete Journal Entry</DialogTitle>
  1104→          </DialogHeader>
  1105→          <p className="text-xs text-muted-foreground">
  1106→            Are you sure you want to delete this journal entry? This action cannot be undone.
  1107→          </p>
  1108→          <div className="flex gap-2 mt-3">
  1109→            <Button
  1110→              variant="outline"
  1111→              size="sm"
  1112→              className="flex-1 h-8 text-xs"
  1113→              onClick={() => setDeleteConfirm(null)}
  1114→            >
  1115→              Cancel
  1116→            </Button>
  1117→            <Button
  1118→              variant="destructive"
  1119→              size="sm"
  1120→              className="flex-1 h-8 text-xs"
  1121→              onClick={() => deleteConfirm && handleDelete(deleteConfirm)}
  1122→            >
  1123→              Delete
  1124→            </Button>
  1125→          </div>
  1126→        </DialogContent>
  1127→      </Dialog>
  1128→    </div>
  1129→  );
  1130→}
