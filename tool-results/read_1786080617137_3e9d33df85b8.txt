    20→    20→// Deterministic pseudo-random number generator (stable across renders)
    21→    21→function seededRandom(seed: number) {
    22→    22→  let s = seed;
    23→    23→  return () => {
    24→    24→    s = (s * 16807 + 0) % 2147483647;
    25→    25→    return (s - 1) / 2147483646;
    26→    26→  };
    27→    27→}

    40→    40→function getSessionStatus(session: { start: number; end: number }) {
    41→    41→  const now = new Date();
    42→    42→  const currentHour = now.getUTCHours();
    43→    43→  const currentMinute = now.getUTCMinutes();
    44→    44→  const currentDecimal = currentHour + currentMinute / 60;
    45→    45→  const isActive = currentDecimal >= session.start && currentDecimal < session.end;
    46→    46→  const progress = isActive
    47→    47→    ? ((currentDecimal - session.start) / (session.end - session.start)) * 100
    48→    48→    : 0;
    49→    49→  const duration = session.end - session.start;
    50→    50→  let statusText = '';
    51→    51→  if (isActive) {
    52→    52→    statusText = 'ACTIVE';
    53→    53→  } else if (currentDecimal < session.start) {
    54→    54→    const hoursUntil = session.start - currentHour;
    55→    55→    const minsUntil = (hoursUntil * 60 - currentMinute);
    56→    56→    if (minsUntil > 0) {
    57→    57→      const h = Math.floor(minsUntil / 60);
    58→    58→      const m = minsUntil % 60;
    59→    59→      statusText = `Opens in ${h}:${m.toString().padStart(2, '0')}`;
    60→    60→    } else {
    61→    61→      statusText = 'Closed';
    62→    62→    }
    63→    63→  } else {
    64→    64→    statusText = 'Closed';
    65→    65→  }
    66→    66→  return { isActive, progress, statusText, duration };
    67→    67→}

    69→    69→function timeAgo(dateStr: string): string {
    70→    70→  const now = Date.now();
    71→    71→  const then = new Date(dateStr).getTime();
    72→    72→  const diffMs = now - then;
    73→    73→  const diffSec = Math.floor(diffMs / 1000);
    74→    74→  if (diffSec < 60) return `${diffSec}s ago`;
    75→    75→  const diffMin = Math.floor(diffSec / 60);
    76→    76→  if (diffMin < 60) return `${diffMin}m ago`;
    77→    77→  const diffHr = Math.floor(diffMin / 60);
    78→    78→  return `${diffHr}h ago`;
    79→    79→}

    81→    81→function MiniSparkline({ values, width = 48, height = 18 }: { values: number[]; width?: number; height?: number }) {
    82→    82→  if (values.length < 2) return null;
    83→    83→  const min = Math.min(...values);
    84→    84→  const max = Math.max(...values);
    85→    85→  const range = max - min || 1;
    86→    86→  const padding = 2;
    87→    87→  const points = values.map((v, i) => {
    88→    88→    const x = padding + (i / (values.length - 1)) * (width - padding * 2);
    89→    89→    const y = padding + (1 - (v - min) / range) * (height - padding * 2);
    90→    90→    return `${x},${y}`;
    91→    91→  }).join(' ');
    92→    92→  const isUp = values[values.length - 1] >= values[0];
    93→    93→  const color = isUp ? '#10b981' : '#ef4444';
    94→    94→  return (
    95→    95→    <svg width={width} height={height} className="inline-block">
    96→    96→      <polyline
    97→    97→        fill="none"
    98→    98→        stroke={color}
    99→    99→        strokeWidth={1.5}
   100→   100→        strokeLinecap="round"
   101→   101→        strokeLinejoin="round"
   102→   102→        points={points}
   103→   103→      />
   104→   104→    </svg>
   105→   105→  );
   106→   106→}
