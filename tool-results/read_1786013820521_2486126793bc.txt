    65→function ConfidenceGauge({ confidence, size = 48 }: { confidence: number; size?: number }) {
    66→  const strokeWidth = 4;
    67→  const radius = (size - strokeWidth) / 2;
    68→  const circumference = Math.PI * radius; // half circle
    69→  const progress = (confidence / 100) * circumference;
    70→  const color = confidence >= 80 ? '#10b981' : confidence >= 60 ? '#f59e0b' : '#ef4444';
    71→  return (
    72→    <svg width={size} height={size / 2 + strokeWidth} viewBox={`0 0 ${size} ${size / 2 + strokeWidth}`} className="overflow-visible">
    73→      {/* background arc */}
    74→      <path
    75→        d={`M ${strokeWidth / 2} ${size / 2} A ${radius} ${radius} 0 0 1 ${size - strokeWidth / 2} ${size / 2}`}
    76→        fill="none"
    77→        stroke="currentColor"
    78→        strokeWidth={strokeWidth}
    79→        className="text-slate-700"
    80→        strokeLinecap="round"
    81→      />
    82→      {/* progress arc */}
    83→      <path
    84→        d={`M ${strokeWidth / 2} ${size / 2} A ${radius} ${radius} 0 0 1 ${size - strokeWidth / 2} ${size / 2}`}
    85→        fill="none"
    86→        stroke={color}
    87→        strokeWidth={strokeWidth}
    88→        strokeLinecap="round"
    89→        strokeDasharray={circumference}
    90→        strokeDashoffset={circumference - progress}
    91→      />
    92→    </svg>
    93→  );
    94→}

    96→function timeAgo(dateStr: string): string {
    97→  const now = Date.now();
    98→  const then = new Date(dateStr).getTime();
    99→  const diff = Math.floor((now - then) / 1000);
   100→  if (diff < 60) return `${diff}s ago`;
   101→  if (diff < 3600) return `${Math.floor(diff / 60)}m ago`;
   102→  if (diff < 86400) return `${Math.floor(diff / 3600)}h ago`;
   103→  return `${Math.floor(diff / 86400)}d ago`;
   104→}

   129→function calculateCorrelation(prices1: number[], prices2: number[]): number {
   130→  if (prices1.length !== prices2.length || prices1.length < 10) return 0;
   131→  const n = prices1.length;
   132→  const mean1 = prices1.reduce((a, b) => a + b, 0) / n;
   133→  const mean2 = prices2.reduce((a, b) => a + b, 0) / n;
   134→  let cov = 0, var1 = 0, var2 = 0;
   135→  for (let i = 0; i < n; i++) {
   136→    const d1 = prices1[i] - mean1;
   137→    const d2 = prices2[i] - mean2;
   138→    cov += d1 * d2;
   139→    var1 += d1 * d1;
   140→    var2 += d2 * d2;
   141→  }
   142→  if (var1 === 0 || var2 === 0) return 0;
   143→  return cov / Math.sqrt(var1 * var2);
   144→}

   150→  if (r < -0.2) return 'bg-red-500/10 text-red-500/70';
   151→  return 'bg-slate-500/10 text-slate-400';
   152→}

   154→function CorrelationGrid({ priceHistory }: { priceHistory: Record<Symbol, any[]> }) {
   155→  const matrix = useMemo(() => {
   156→    const closePrices: Record<string, number[]> = {};
   157→    SYMBOLS.forEach((sym) => {
   158→      const history = priceHistory[sym as Symbol];
   159→      if (history && history.length >= 10) {
   160→        closePrices[sym] = history.slice(-50).map((c: any) => c.close);
   161→      } else {
   162→        closePrices[sym] = [];
   163→      }
   164→    });
   165→
   166→    const grid: number[][] = [];
   167→    for (let i = 0; i < SYMBOLS.length; i++) {
   168→      const row: number[] = [];
   169→      for (let j = 0; j < SYMBOLS.length; j++) {
   170→        if (i === j) {
   171→          row.push(1);
   172→        } else {
   173→          const p1 = closePrices[SYMBOLS[i]];
   174→          const p2 = closePrices[SYMBOLS[j]];
   175→          if (p1.length >= 10 && p2.length >= 10) {
   176→            const len = Math.min(p1.length, p2.length);
   177→            row.push(calculateCorrelation(p1.slice(-len), p2.slice(-len)));
   178→          } else {
   179→            row.push(0);
   180→          }
   181→        }
   182→      }
   183→      grid.push(row);
   184→    }
   185→    return grid;
   186→  }, [priceHistory]);
   187→
   188→  return (
   189→    <div className="flex flex-col items-center gap-1">
   190→      {/* Column headers */}
   191→      <div className="flex items-center gap-1 pl-16">
   192→        {SYMBOLS.map((sym) => (
   193→          <div key={sym} className="w-16 h-6 flex items-center justify-center">
   194→            <span className="text-[10px] font-medium text-muted-foreground">{sym}</span>
   195→          </div>
   196→        ))}
   197→      </div>
   198→      {/* Rows */}
   199→      {SYMBOLS.map((sym, i) => (
   200→        <div key={sym} className="flex items-center gap-1">
   201→          <div className="w-16 h-12 flex items-center justify-end pr-2">
   202→            <span className="text-[10px] font-medium text-muted-foreground">{sym}</span>
   203→          </div>
   204→          {SYMBOLS.map((_, j) => {
   205→            const r = matrix[i]?.[j] ?? 0;
   206→            const isDiag = i === j;
   207→            return (
   208→              <div
   209→                key={`${i}-${j}`}
   210→                className={`w-16 h-12 rounded-md flex items-center justify-center ${
   211→                  isDiag ? 'bg-slate-700/50 text-foreground font-semibold' : getCorrelationColor(r)
   212→                }`}
   213→              >
   214→                <span className="text-[10px] tabular-nums">{r.toFixed(2)}</span>
   215→              </div>
   216→            );
   217→          })}
   218→        </div>
   219→      ))}
   220→    </div>
   221→  );
   222→}
