    18→// Seed-based pseudo-random for consistent mock data
    19→function seededRandom(seed: number) {
    20→  let s = seed;
    21→  return () => {
    22→    s = (s * 16807 + 0) % 2147483647;
    23→    return (s - 1) / 2147483646;
    24→  };
    25→}

    27→function generateMockEquityCurve(days: number) {
    28→  const rng = seededRandom(42);
    29→  const data: { date: string; equity: number; peak: number; drawdown: number }[] = [];
    30→  let equity = 10000;
    31→  let peak = 10000;
    32→  const now = new Date();
    33→  for (let i = days - 1; i >= 0; i--) {
    34→    const d = new Date(now);
    35→    d.setDate(d.getDate() - i);
    36→    const dayReturn = (rng() - 0.42) * 80; // slight positive bias
    37→    equity = Math.max(9200, equity + dayReturn);
    38→    peak = Math.max(peak, equity);
    39→    const dd = ((peak - equity) / peak) * 100;
    40→    data.push({
    41→      date: d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' }),
    42→      equity: Math.round(equity * 100) / 100,
    43→      peak: Math.round(peak * 100) / 100,
    44→      drawdown: Math.round(dd * 100) / 100,
    45→    });
    46→  }
    47→  return data;
    48→}

    50→function generateMockDailyPnl(days: number) {
    51→  const rng = seededRandom(123);
    52→  const data: { date: string; pnl: number }[] = [];
    53→  const now = new Date();
    54→  for (let i = days - 1; i >= 0; i--) {
    55→    const d = new Date(now);
    56→    d.setDate(d.getDate() - i);
    57→    const pnl = (rng() - 0.42) * 120;
    58→    data.push({
    59→      date: d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' }),
    60→      pnl: Math.round(pnl * 100) / 100,
    61→    });
    62→  }
    63→  return data;
    64→}

    66→function generateMockHeatmap(): { day: string; week: number; pnl: number }[] {
    67→  const rng = seededRandom(777);
    68→  const days = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
    69→  const data: { day: string; week: number; pnl: number }[] = [];
    70→  for (let w = 0; w < 5; w++) {
    71→    for (const day of days) {
    72→      if (day === 'Sat' || day === 'Sun') {
    73→        data.push({ day, week: w, pnl: 0 });
    74→      } else {
    75→        const pnl = (rng() - 0.4) * 200;
    76→        data.push({ day, week: w, pnl: Math.round(pnl) });
    77→      }
    78→    }
    79→  }
    80→  return data;
    81→}
