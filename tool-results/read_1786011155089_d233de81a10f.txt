    46→    46→function generateMockBacktestResult(strategy: StrategyName, symbol: Symbol): BacktestResult {
    47→    47→  const initialBalance = 10000;
    48→    48→  const totalTrades = Math.floor(Math.random() * 150) + 50;
    49→    49→  const winRate = 40 + Math.random() * 30;
    50→    50→  const wins = Math.floor(totalTrades * winRate / 100);
    51→    51→  const losses = totalTrades - wins;
    52→    52→  const avgWin = 15 + Math.random() * 30;
    53→    53→  const avgLoss = 8 + Math.random() * 15;
    54→    54→  const totalProfit = wins * avgWin;
    55→    55→  const totalLoss = losses * avgLoss;
    56→    56→  const finalBalance = initialBalance + totalProfit - totalLoss;
    57→    57→  const maxDD = 5 + Math.random() * 20;
    58→    58→  const sharpeRatio = 0.5 + Math.random() * 2.5;
    59→    59→  const profitFactor = totalLoss > 0 ? totalProfit / totalLoss : 0;
    60→    60→
    61→    61→  const equityCurve: { trade: number; equity: number }[] = [];
    62→    62→  let equity = initialBalance;
    63→    63→  for (let i = 0; i < totalTrades; i++) {
    64→    64→    const isWin = Math.random() * 100 < winRate;
    65→    65→    equity += isWin ? avgWin * (0.5 + Math.random()) : -avgLoss * (0.5 + Math.random());
    66→    66→    equityCurve.push({ trade: i + 1, equity: Math.round(equity * 100) / 100 });
    67→    67→  }
    68→    68→
    69→    69→  return {
    70→    70→    id: `bt-${Date.now()}`,
    71→    71→    name: STRATEGIES[strategy].label,
    72→    72→    symbol,
    73→    73→    strategy,
    74→    74→    initialBalance,
    75→    75→    finalBalance: Math.round(finalBalance * 100) / 100,
    76→    76→    totalTrades,
    77→    77→    winRate: Math.round(winRate * 10) / 10,
    78→    78→    profitFactor: Math.round(profitFactor * 100) / 100,
    79→    79→    maxDrawdown: Math.round(maxDD * 10) / 10,
    80→    80→    sharpeRatio: Math.round(sharpeRatio * 100) / 100,
    81→    81→    totalProfit: Math.round(totalProfit * 100) / 100,
    82→    82→    totalLoss: Math.round(totalLoss * 100) / 100,
    83→    83→    avgWin: Math.round(avgWin * 100) / 100,
    84→    84→    avgLoss: Math.round(avgLoss * 100) / 100,
    85→    85→    equityCurve,
    86→    86→  };
    87→    87→}

    89→    89→function WinRateRing({ value }: { value: number }) {
    90→    90→  const radius = 20;
    91→    91→  const circumference = 2 * Math.PI * radius;
    92→    92→  const offset = circumference * (1 - value / 100);
    93→    93→  const color = value >= 55 ? '#10b981' : value >= 45 ? '#f59e0b' : '#ef4444';
    94→    94→
    95→    95→  return (
    96→    96→    <div className="relative w-12 h-12 flex-shrink-0">
    97→    97→      <svg width="48" height="48" className="-rotate-90">
    98→    98→        <circle cx="24" cy="24" r={radius} fill="none" stroke="rgba(255,255,255,0.06)" strokeWidth="3.5" />
    99→    99→        <circle
   100→   100→          cx="24" cy="24" r={radius} fill="none" stroke={color} strokeWidth="3.5"
   101→   101→          strokeDasharray={circumference} strokeDashoffset={offset} strokeLinecap="round"
   102→   102→          style={{ transition: 'stroke-dashoffset 0.6s ease' }}
   103→   103→        />
   104→   104→      </svg>
   105→   105→      <div className="absolute inset-0 flex items-center justify-center">
   106→   106→        <span className="text-[10px] font-bold tabular-nums" style={{ color }}>{Math.round(value)}</span>
   107→   107→      </div>
   108→   108→    </div>
   109→   109→  );
   110→   110→}
