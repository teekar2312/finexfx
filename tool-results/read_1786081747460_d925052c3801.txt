    89→function formatPrice(price: number, symbol: string) {
    90→  if (symbol === 'XAUUSD') return price.toFixed(2);
    91→  if (symbol === 'USDJPY') return price.toFixed(3);
    92→  return price.toFixed(5);
    93→}

    95→function formatDate(dateStr: string) {
    96→  const d = new Date(dateStr);
    97→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    98→}
    99→
   100→function formatTime(dateStr: string) {
   101→  const d = new Date(dateStr);
   102→  return d.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', hour12: false });
   103→}

   125→// --- Journal Entry Form ---

   129→  onCancel,

   131→  entry?: JournalEntry;

   135→  const isEdit = !!entry;
   136→  const [symbol, setSymbol] = useState(entry?.symbol || 'EURUSD');
   137→  const [direction, setDirection] = useState<'BUY' | 'SELL'>(entry?.direction || 'BUY');
   138→  const [entryPrice, setEntryPrice] = useState(entry?.entryPrice.toString() || '');
   139→  const [exitPrice, setExitPrice] = useState(entry?.exitPrice.toString() || '');
   140→  const [lotSize, setLotSize] = useState(entry?.lotSize.toString() || '0.1');
   141→  const [pips, setPips] = useState(entry?.pips.toString() || '');
   142→  const [pnl, setPnl] = useState(entry?.pnl.toString() || '');
   143→  const [strategy, setStrategy] = useState(entry?.strategy || 'EMA Crossover');
   144→  const [openTime, setOpenTime] = useState(entry?.openTime ? entry.openTime.slice(0, 16) : '');
   145→  const [closeTime, setCloseTime] = useState(entry?.closeTime ? entry.closeTime.slice(0, 16) : '');
   146→  const [duration, setDuration] = useState(entry?.duration || '');
   147→  const [notes, setNotes] = useState(entry?.notes || '');
   148→  const [tagsStr, setTagsStr] = useState(entry?.tags.join(', ') || '');
   149→  const [mood, setMood] = useState<JournalEntry['mood']>(entry?.mood || 'neutral');
   150→  const [mistakesStr, setMistakesStr] = useState(entry?.mistakes.join(', ') || '');
   151→  const [lessons, setLessons] = useState(entry?.lessons || '');
   152→  const [rating, setRating] = useState(entry?.rating || 3);
   153→
   154→  const handleSubmit = (e: React.FormEvent) => {
   155→    e.preventDefault();
   156→    onSave({
   157→      tradeId: entry?.tradeId || `t-${Date.now()}`,
   158→      symbol,
   159→      direction,
   160→      entryPrice: parseFloat(entryPrice) || 0,
   161→      exitPrice: parseFloat(exitPrice) || 0,
   162→      pips: parseFloat(pips) || 0,
   163→      pnl: parseFloat(pnl) || 0,
   164→      lotSize: parseFloat(lotSize) || 0.01,
   165→      strategy,
   166→      openTime: openTime ? new Date(openTime).toISOString() : new Date().toISOString(),
   167→      closeTime: closeTime ? new Date(closeTime).toISOString() : new Date().toISOString(),
   168→      duration: duration || 'N/A',
   169→      notes,
   170→      tags: tagsStr.split(',').map((t) => t.trim()).filter(Boolean),
   171→      mood,
   172→      mistakes: mistakesStr.split(',').map((m) => m.trim()).filter(Boolean),
   173→      lessons,
   174→      rating,
   175→    });
   176→  };
   177→
   178→  return (
   179→    <form onSubmit={handleSubmit} className="space-y-4">
   180→      <div className="grid grid-cols-2 gap-3">
   181→        <div className="space-y-1.5">
   182→          <Label className="text-xs text-muted-foreground">Symbol</Label>
   183→          <Select value={symbol} onValueChange={setSymbol}>
   184→            <SelectTrigger className="h-9 text-xs"><SelectValue /></SelectTrigger>
   185→            <SelectContent>
   186→              {SYMBOLS_LIST.map((s) => (
   187→                <SelectItem key={s} value={s} className="text-xs">{SYMBOL_DISPLAY[s]}</SelectItem>
   188→              ))}
   189→            </SelectContent>
   190→          </Select>
   191→        </div>
   192→        <div className="space-y-1.5">
   193→          <Label className="text-xs text-muted-foreground">Direction</Label>
   194→          <div className="flex gap-2">
   195→            <Button
   196→              type="button"
   197→              size="sm"
   198→              variant={direction === 'BUY' ? 'default' : 'outline'}
   199→              className={`flex-1 h-9 text-xs ${direction === 'BUY' ? 'bg-emerald-600 hover:bg-emerald-700' : ''}`}
   200→              onClick={() => setDirection('BUY')}
   201→            >BUY</Button>
   202→            <Button
   203→              type="button"
   204→              size="sm"
   205→              variant={direction === 'SELL' ? 'default' : 'outline'}
   206→              className={`flex-1 h-9 text-xs ${direction === 'SELL' ? 'bg-red-600 hover:bg-red-700' : ''}`}
   207→              onClick={() => setDirection('SELL')}
   208→            >SELL</Button>
   209→          </div>
   210→        </div>
   211→      </div>
   212→
   213→      <div className="grid grid-cols-2 gap-3">
   214→        <div className="space-y-1.5">
   215→          <Label className="text-xs text-muted-foreground">Entry Price</Label>
   216→          <Input type="number" step="any" value={entryPrice} onChange={(e) => setEntryPrice(e.target.value)} className="h-9 text-xs" />

   219→          <Label className="text-xs text-muted-foreground">Exit Price</Label>
   220→          <Input type="number" step="any" value={exitPrice} onChange={(e) => setExitPrice(e.target.value)} className="h-9 text-xs" />
   221→        </div>
   222→      </div>
   223→
   224→      <div className="grid grid-cols-3 gap-3">
   225→        <div className="space-y-1.5">
   226→          <Label className="text-xs text-muted-foreground">Pips</Label>
   227→          <Input type="number" step="any" value={pips} onChange={(e) => setPips(e.target.value)} className="h-9 text-xs" />
   228→        </div>
   229→        <div className="space-y-1.5">
   230→          <Label className="text-xs text-muted-foreground">P&L ($)</Label>
   231→          <Input type="number" step="any" value={pnl} onChange={(e) => setPnl(e.target.value)} className="h-9 text-xs" />
   232→        </div>
   233→        <div className="space-y-1.5">
   234→          <Label className="text-xs text-muted-foreground">Lot Size</Label>
   235→          <Input type="number" step="0.01" value={lotSize} onChange={(e) => setLotSize(e.target.value)} className="h-9 text-xs" />
   236→        </div>
   237→      </div>
   238→
   239→      <div className="grid grid-cols-2 gap-3">
   240→        <div className="space-y-1.5">

   242→          <Select value={strategy} onValueChange={setStrategy}>
   243→            <SelectTrigger className="h-9 text-xs"><SelectValue /></SelectTrigger>
   244→            <SelectContent>
   245→              {STRATEGIES.map((s) => (
   246→                <SelectItem key={s} value={s} className="text-xs">{s}</SelectItem>

   248→            </SelectContent>
   249→          </Select>
   250→        </div>
   251→        <div className="space-y-1.5">
   252→          <Label className="text-xs text-muted-foreground">Duration</Label>
   253→          <Input value={duration} onChange={(e) => setDuration(e.target.value)} placeholder="e.g. 1h 30m" className="h-9 text-xs" />
   254→        </div>

   256→
   257→      <div className="grid grid-cols-2 gap-3">
   258→        <div className="space-y-1.5">

   263→          <Label className="text-xs text-muted-foreground">Close Time</Label>

   265→        </div>
   266→      </div>

   268→      <div className="space-y-1.5">
   269→        <Label className="text-xs text-muted-foreground">Tags (comma-separated)</Label>

   274→        <Label className="text-xs text-muted-foreground">Trade Notes</Label>

   278→      <div className="space-y-2">
   279→        <Label className="text-xs text-muted-foreground">Mood</Label>
   280→        <div className="flex gap-2">
   281→          {(Object.keys(MOOD_CONFIG) as Array<JournalEntry['mood']>).map((m) => {
   282→            const cfg = MOOD_CONFIG[m];
   283→            return (
   284→              <button
   285→                key={m}
   286→                type="button"
   287→                onClick={() => setMood(m)}
   288→                className={`flex-1 flex flex-col items-center gap-1 p-2 rounded-lg border transition-all text-xs

   299→      <div className="space-y-1.5">
   300→        <Label className="text-xs text-muted-foreground">Mistakes (comma-separated)</Label>

   326→// --- Journal Entry Card ---
   327→function JournalEntryCard({
   328→  entry,
   329→  index,
   330→  onEdit,
   331→  onDelete,
   332→}: {
   333→  entry: JournalEntry;
   334→  index: number;
   335→  onEdit: (entry: JournalEntry) => void;
   336→  onDelete: (id: string) => void;
   337→}) {
   338→  const [expanded, setExpanded] = useState(false);
   339→  const isWin = entry.pnl >= 0;
   340→  const moodCfg = MOOD_CONFIG[entry.mood];
   341→
   342→  return (
   343→    <motion.div
   344→      initial={{ opacity: 0, y: 8 }}
   345→      animate={{ opacity: 1, y: 0 }}
   346→      transition={{ delay: index * 0.04, duration: 0.2 }}
   347→      className="glass-card rounded-lg overflow-hidden parallax-hover"
   348→    >
   349→      {/* Card Header */}
   350→      <div className="p-4">
   351→        <div className="flex items-start justify-between gap-3">
   352→          <div className="flex items-center gap-3 min-w-0">

   354→            <div className={`flex-shrink-0 w-10 h-10 rounded-lg flex items-center justify-center
   355→              ${entry.direction === 'BUY' ? 'bg-emerald-500/10' : 'bg-red-500/10'}`}>

   357→                ? <ArrowUpRight className={`h-5 w-5 text-emerald-400`} />
   358→                : <ArrowDownRight className={`h-5 w-5 text-red-400`} />
   359→              }
   360→            </div>
   361→
   362→            <div className="min-w-0 flex-1">
   363→              <div className="flex items-center gap-2 flex-wrap">
   364→                <span className="text-sm font-semibold">{SYMBOL_DISPLAY[entry.symbol] || entry.symbol}</span>
   365→                <Badge
   366→                  variant="outline"
   367→                  className={`text-[10px] px-1.5 py-0 font-semibold
   368→                    ${entry.direction === 'BUY' ? 'badge-glow-emerald border-emerald-500/40 text-emerald-400' : 'badge-glow-red border-red-500/40 text-red-400'}`}
   369→                >
   370→                  {entry.direction}
   371→                </Badge>
   372→                <Badge variant="outline" className="text-[10px] px-1.5 py-0 text-muted-foreground">
   373→                  {entry.strategy}
   374→                </Badge>
   375→              </div>
   376→              <div className="flex items-center gap-3 mt-1 text-[11px] text-muted-foreground">
   377→                <span className="flex items-center gap-1">
   378→                  <CalendarDays className="h-3 w-3" />

   381→                <span className="flex items-center gap-1">
   382→                  <Clock className="h-3 w-3" />
   383→                  {entry.duration}
   384→                </span>
   385→              </div>
   386→            </div>
   387→          </div>
   388→
   389→          <div className="flex flex-col items-end gap-1 flex-shrink-0">
   390→            <span className={`text-sm font-bold tabular-nums ${isWin ? 'text-emerald-400 neon-text-emerald' : 'text-red-400 neon-text-red'}`}>
   391→              {isWin ? '+' : ''}{entry.pnl.toFixed(2)}
   392→            </span>
   393→            <span className={`text-[11px] tabular-nums ${isWin ? 'text-emerald-400/70' : 'text-red-400/70'}`}>
   394→              {isWin ? '+' : ''}{entry.pips} pips
   395→            </span>
   396→          </div>
   397→        </div>
   398→
   399→        {/* Price info row */}
   400→        <div className="flex items-center gap-4 mt-3 text-[11px] text-muted-foreground">
   401→          <span>Entry: <span className="text-foreground font-medium tabular-nums">{formatPrice(entry.entryPrice, entry.symbol)}</span></span>

   406→
   407→        {/* Mood, rating, tags row */}
   408→        <div className="flex items-center gap-3 mt-3 flex-wrap">
   409→          <div className={`flex items-center gap-1.5 px-2 py-0.5 rounded-full text-[10px] font-medium ${moodCfg.bg} ${moodCfg.color} border ${moodCfg.border}`}>
   410→            <span>{moodCfg.emoji}</span>
   411→            <span>{moodCfg.label}</span>
   412→          </div>
   413→          <StarRating rating={entry.rating} />
   414→          <div className="flex items-center gap-1 ml-auto flex-wrap">
   415→            {entry.tags.slice(0, 3).map((tag) => (
   416→              <Badge key={tag} variant="secondary" className="text-[9px] px-1.5 py-0 h-5 font-normal">
   417→                {tag}
   418→              </Badge>
   419→            ))}
   420→            {entry.tags.length > 3 && (
   421→              <span className="text-[10px] text-muted-foreground">+{entry.tags.length - 3}</span>
   422→            )}
   423→          </div>
   424→        </div>
   425→
   426→        {/* Preview of notes */
   427→          !expanded && entry.notes && (
   428→            <p className="text-[11px] text-muted-foreground mt-3 line-clamp-2 leading-relaxed">
   429→              {entry.notes}
   430→            </p>
   431→          )
   432→        }
   433→      </div>
   434→
   435→      {/* Expand/Collapse Button */

   437→          onClick={() => setExpanded(!expanded)}
   438→          className="w-full flex items-center justify-center gap-1 py-2 text-[11px] text-muted-foreground hover:text-foreground hover:bg-accent/50 transition-colors border-t border-border/50"

   440→          {expanded ? (
   441→            <><ChevronUp className="h-3 w-3" /> Show less</>
   442→          ) : (
   443→            <><ChevronDown className="h-3 w-3" /> Show details</>
   444→          )}
   445→        </button>
   446→      }
   447→
   448→      {/* Expanded Content */
   449→        <AnimatePresence>
   450→          {expanded && (
   451→            <motion.div
   452→              initial={{ height: 0, opacity: 0 }}
   453→              animate={{ height: 'auto', opacity: 1 }}
   454→              exit={{ height: 0, opacity: 0 }}
   455→              transition={{ duration: 0.2 }}
   456→              className="overflow-hidden"
   457→            >
   458→              <div className="px-4 pb-4 space-y-4">
   459→                <Separator />
   460→
   461→                {/* Full Notes */}
   462→                {entry.notes && (
   463→                  <div>
   464→                    <div className="flex items-center gap-1.5 mb-1.5">
   465→                      <BookOpen className="h-3 w-3 text-muted-foreground" />
   466→                      <span className="text-[11px] font-medium text-muted-foreground uppercase tracking-wider">Notes</span>
   467→                    </div>
   468→                    <p className="text-xs text-foreground/80 leading-relaxed bg-accent/30 rounded-lg p-3">
   469→                      {entry.notes}
   470→                    </p>
   471→                  </div>
   472→                )}
   473→
   474→                {/* Mistakes */}
   475→                {entry.mistakes.length > 0 && (

   563→function JournalAnalytics({ entries }: { entries: JournalEntry[] }) {
   564→  const wins = entries.filter((e) => e.pnl >= 0);
   565→  const losses = entries.filter((e) => e.pnl < 0);
   566→  const winRate = entries.length > 0 ? (wins.length / entries.length) * 100 : 0;
   567→  const totalPnl = entries.reduce((s, e) => s + e.pnl, 0);
   568→  const avgPnl = entries.length > 0 ? totalPnl / entries.length : 0;
   569→  const avgRating = entries.length > 0 ? entries.reduce((s, e) => s + e.rating, 0) / entries.length : 0;

   571→  const avgLoss = losses.length > 0 ? losses.reduce((s, e) => s + e.pnl, 0) / losses.length : 0;
   572→
   573→  // Strategy performance
   574→  const strategyMap = new Map<string, { count: number; pnl: number; wins: number }>();
   575→  entries.forEach((e) => {
   576→    const s = strategyMap.get(e.strategy) || { count: 0, pnl: 0, wins: 0 };
   577→    s.count++;
   578→    s.pnl += e.pnl;
   579→    if (e.pnl >= 0) s.wins++;
   580→    strategyMap.set(e.strategy, s);
