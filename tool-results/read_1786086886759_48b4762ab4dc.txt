    59→function generateMockBacktestResult(strategy: StrategyName, symbol: Symbol): BacktestResult {
    60→  const initialBalance = 10000;
    61→  const totalTrades = Math.floor(Math.random() * 150) + 50;
    62→  const winRate = 40 + Math.random() * 30;
    63→  const wins = Math.floor(totalTrades * winRate / 100);
    64→  const losses = totalTrades - wins;
    65→  const avgWin = 15 + Math.random() * 30;
    66→  const avgLoss = 8 + Math.random() * 15;
    67→  const totalProfit = wins * avgWin;
    68→  const totalLoss = losses * avgLoss;
    69→  const finalBalance = initialBalance + totalProfit - totalLoss;
    70→  const maxDD = 5 + Math.random() * 20;
    71→  const sharpeRatio = 0.5 + Math.random() * 2.5;
    72→  const profitFactor = totalLoss > 0 ? totalProfit / totalLoss : 0;
    73→
    74→  const equityCurve: { trade: number; equity: number }[] = [];
    75→  let equity = initialBalance;
    76→  for (let i = 0; i < totalTrades; i++) {
    77→    const isWin = Math.random() * 100 < winRate;
    78→    equity += isWin ? avgWin * (0.5 + Math.random()) : -avgLoss * (0.5 + Math.random());
    79→    equityCurve.push({ trade: i + 1, equity: Math.round(equity * 100) / 100 });
    80→  }
    81→
    82→  return {
    83→    id: `bt-${Date.now()}`,
    84→    name: STRATEGIES[strategy].label,
    85→    symbol,
    86→    strategy,
    87→    initialBalance,
    88→    finalBalance: Math.round(finalBalance * 100) / 100,
    89→    totalTrades,
    90→    winRate: Math.round(winRate * 10) / 10,
    91→    profitFactor: Math.round(profitFactor * 100) / 100,
    92→    maxDrawdown: Math.round(maxDD * 10) / 10,
    93→    sharpeRatio: Math.round(sharpeRatio * 100) / 100,
    94→    totalProfit: Math.round(totalProfit * 100) / 100,
    95→    totalLoss: Math.round(totalLoss * 100) / 100,
    96→    avgWin: Math.round(avgWin * 100) / 100,
    97→    avgLoss: Math.round(avgLoss * 100) / 100,
    98→    equityCurve,
    99→  };
   100→}

   102→function WinRateRing({ value }: { value: number }) {
   103→  const radius = 20;
   104→  const circumference = 2 * Math.PI * radius;
   105→  const offset = circumference * (1 - value / 100);
   106→  const color = value >= 55 ? '#10b981' : value >= 45 ? '#f59e0b' : '#ef4444';
   107→
   108→  return (
   109→    <div className="relative w-12 h-12 flex-shrink-0">
   110→      <svg width="48" height="48" className="-rotate-90">
   111→        <circle cx="24" cy="24" r={radius} fill="none" stroke="rgba(255,255,255,0.06)" strokeWidth="3.5" />
   112→        <circle
   113→          cx="24" cy="24" r={radius} fill="none" stroke={color} strokeWidth="3.5"
   114→          strokeDasharray={circumference} strokeDashoffset={offset} strokeLinecap="round"
   115→          style={{ transition: 'stroke-dashoffset 0.6s ease' }}
   116→        />
   117→      </svg>
   118→      <div className="absolute inset-0 flex items-center justify-center">
   119→        <span className="text-[10px] font-bold tabular-nums" style={{ color }}>{Math.round(value)}</span>
   120→      </div>
   121→    </div>
   122→  );
   123→}
