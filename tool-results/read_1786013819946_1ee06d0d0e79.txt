    62→function ImpactDots({ impact }: { impact: 'high' | 'medium' | 'low' }) {
    63→  const count = impact === 'high' ? 3 : impact === 'medium' ? 2 : 1;
    64→  const color = impact === 'high' ? 'bg-red-500' : impact === 'medium' ? 'bg-amber-500' : 'bg-green-500';
    65→  return (
    66→    <div className="flex items-center gap-0.5">
    67→      {Array.from({ length: 3 }).map((_, i) => (
    68→        <div key={i} className={`w-1.5 h-1.5 rounded-full ${i < count ? color : 'bg-slate-700'}`} />
    69→      ))}
    70→    </div>
    71→  );
    72→}

    74→function getActualVsForecast(actual: string, forecast: string): 'better' | 'worse' | 'same' {
    75→  const a = parseFloat(actual.replace('%', '').replace('K', ''));
    76→  const f = parseFloat(forecast.replace('%', '').replace('K', ''));
    77→  if (isNaN(a) || isNaN(f)) return 'same';
    78→  if (a > f) return 'better';
    79→  if (a < f) return 'worse';
    80→  return 'same';
    81→}
    82→
    83→function getActualColor(actual: string, forecast: string): string {
    84→  const comparison = getActualVsForecast(actual, forecast);
    85→  return comparison === 'better' ? 'text-emerald-500' : comparison === 'worse' ? 'text-red-500' : 'text-foreground';
    86→}
    87→
    88→function ActualVsForecastArrow({ actual, forecast }: { actual: string; forecast: string }) {
    89→  const comparison = getActualVsForecast(actual, forecast);
    90→  if (comparison === 'better') return <ArrowUpRight className="h-3 w-3 text-emerald-500" />;
    91→  if (comparison === 'worse') return <ArrowDownRight className="h-3 w-3 text-red-500" />;
    92→  return <Minus className="h-3 w-3 text-slate-500" />;
    93→}

    95→function timeAgo(dateStr: string): string {
    96→  const now = Date.now();
    97→  const then = new Date(dateStr).getTime();
    98→  const diff = Math.floor((now - then) / 1000);
    99→  if (diff < 60) return `${diff}s ago`;
   100→  if (diff < 3600) return `${Math.floor(diff / 60)}m ago`;
   101→  if (diff < 86400) return `${Math.floor(diff / 3600)}h ago`;
   102→  return `${Math.floor(diff / 86400)}d ago`;
   103→}
