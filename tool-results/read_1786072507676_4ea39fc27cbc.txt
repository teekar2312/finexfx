    16→function DonutGauge({ value, max, size = 80, strokeWidth = 6, showPercent = false }: { value: number; max: number; size?: number; strokeWidth?: number; showPercent?: boolean }) {
    17→  const radius = (size - strokeWidth) / 2;
    18→  const circumference = 2 * Math.PI * radius;
    19→  const ratio = Math.min(value / max, 1);
    20→  const offset = circumference * (1 - ratio);
    21→  const isSafe = ratio < 0.5;
    22→  const isWarn = ratio >= 0.5 && ratio < 0.8;
    23→  const isDanger = ratio >= 0.8;
    24→  const color = isSafe ? '#10b981' : isWarn ? '#f59e0b' : '#ef4444';
    25→  const glowColor = isSafe ? 'rgba(16,185,129,0.35)' : isWarn ? 'rgba(245,158,11,0.35)' : 'rgba(239,68,68,0.35)';
    26→  const pct = Math.round(ratio * 100);
    27→
    28→  return (
    29→    <motion.div
    30→      initial={{ opacity: 0, scale: 0.85 }}
    31→      animate={{ opacity: 1, scale: 1 }}
    32→      transition={{ duration: 0.4, ease: 'easeOut' }}
    33→      className="relative inline-flex items-center justify-center"
    34→    >
    35→      <svg width={size} height={size} className="-rotate-90">
    36→        <defs>
    37→          <filter id={`gauge-glow-${size}`}>
    38→            <feGaussianBlur stdDeviation="3" result="blur" />
    39→            <feMerge><feMergeNode in="blur" /><feMergeNode in="SourceGraphic" /></feMerge>
    40→          </filter>
    41→        </defs>
    42→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="rgba(255,255,255,0.06)" strokeWidth={strokeWidth} />
    43→        {/* Glow layer */}
    44→        <circle
    45→          cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color} strokeWidth={strokeWidth + 4}
    46→          strokeDasharray={circumference} strokeDashoffset={offset} strokeLinecap="round"
    47→          style={{ transition: 'stroke-dashoffset 0.6s ease, stroke 0.3s', opacity: 0.2, filter: `drop-shadow(0 0 6px ${glowColor})` }}
    48→        />
    49→        {/* Main arc */}
    50→        <circle
    51→          cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color} strokeWidth={strokeWidth}
    52→          strokeDasharray={circumference} strokeDashoffset={offset} strokeLinecap="round"
    53→          style={{ transition: 'stroke-dashoffset 0.6s ease, stroke 0.3s' }}
    54→          filter={`url(#gauge-glow-${size})`}
    55→        />
    56→      </svg>
    57→      {showPercent && (
    58→        <div className="absolute inset-0 flex flex-col items-center justify-center rotate-0">
    59→          <span className={`text-sm font-bold tabular-nums ${isSafe ? 'text-emerald-400' : isWarn ? 'text-amber-400' : 'text-red-400'}`}>{pct}%</span>
    60→        </div>
    61→      )}
    62→    </motion.div>
    63→  );
    64→}
