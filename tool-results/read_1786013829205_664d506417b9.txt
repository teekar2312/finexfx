   563→   563→// --- Analytics Section ---
   564→   564→function JournalAnalytics({ entries }: { entries: JournalEntry[] }) {
   565→   565→  const wins = entries.filter((e) => e.pnl >= 0);
   566→   566→  const losses = entries.filter((e) => e.pnl < 0);
   567→   567→  const winRate = entries.length > 0 ? (wins.length / entries.length) * 100 : 0;
   568→   568→  const totalPnl = entries.reduce((s, e) => s + e.pnl, 0);
   569→   569→  const avgPnl = entries.length > 0 ? totalPnl / entries.length : 0;
   570→   570→  const avgRating = entries.length > 0 ? entries.reduce((s, e) => s + e.rating, 0) / entries.length : 0;
   571→   571→  const avgWin = wins.length > 0 ? wins.reduce((s, e) => s + e.pnl, 0) / wins.length : 0;
   572→   572→  const avgLoss = losses.length > 0 ? losses.reduce((s, e) => s + e.pnl, 0) / losses.length : 0;
   573→   573→
   574→   574→  // Strategy performance
   575→   575→  const strategyMap = new Map<string, { count: number; pnl: number; wins: number }>();
   576→   576→  entries.forEach((e) => {
   577→   577→    const s = strategyMap.get(e.strategy) || { count: 0, pnl: 0, wins: 0 };
   578→   578→    s.count++;
   579→   579→    s.pnl += e.pnl;
   580→   580→    if (e.pnl >= 0) s.wins++;
   581→   581→    strategyMap.set(e.strategy, s);
   582→   582→  });
   583→   583→  const strategyData = Array.from(strategyMap.entries()).map(([name, data]) => ({
   584→   584→    name,
   585→   585→    trades: data.count,
   586→   586→    pnl: parseFloat(data.pnl.toFixed(2)),
   587→   587→    winRate: parseFloat(((data.wins / data.count) * 100).toFixed(0)),
   588→   588→  }));
   589→   589→
   590→   590→  // Mood distribution
   591→   591→  const moodMap = new Map<string, number>();
   592→   592→  entries.forEach((e) => moodMap.set(e.mood, (moodMap.get(e.mood) || 0) + 1));
   593→   593→  const moodData = (Object.keys(MOOD_CONFIG) as Array<JournalEntry['mood']>).map((m) => ({
   594→   594→    name: MOOD_CONFIG[m].label,
   595→   595→    value: moodMap.get(m) || 0,
   596→   596→  })).filter((d) => d.value > 0);
   597→   597→
   598→   598→  // P&L by symbol
   599→   599→  const symbolMap = new Map<string, number>();
   600→   600→  entries.forEach((e) => symbolMap.set(e.symbol, (symbolMap.get(e.symbol) || 0) + e.pnl));
   601→   601→  const symbolPnlData = Array.from(symbolMap.entries()).map(([sym, pnl]) => ({
   602→   602→    name: SYMBOL_DISPLAY[sym] || sym,
   603→   603→    pnl: parseFloat(pnl.toFixed(2)),
   604→   604→  }));
   605→   605→
   606→   606→  return (
   607→   607→    <div className="space-y-4 elevated-card rounded-xl p-4">
   608→   608→      {/* Summary Stats */}
   609→   609→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3 stagger-children">
   610→   610→        <Card className="glass-card border-0 p-0 card-hover">
   611→   611→          <CardContent className="p-4">
   612→   612→            <div className="flex items-center gap-2 mb-1">
   613→   613→              <div className="w-7 h-7 rounded-lg bg-primary/10 flex items-center justify-center">
   614→   614→                <Target className="h-3.5 w-3.5 text-primary" />
   615→   615→              </div>
   616→   616→              <span className="text-[10px] text-muted-foreground uppercase tracking-wider">Win Rate</span>
   617→   617→            </div>
   618→   618→            <div className="text-xl font-bold tabular-nums">{winRate.toFixed(0)}%</div>
   619→   619→            <div className="text-[10px] text-muted-foreground mt-0.5">{wins.length}W / {losses.length}L</div>
   620→   620→          </CardContent>
   621→   621→        </Card>
   622→   622→
   623→   623→        <Card className="glass-card border-0 p-0 card-hover">
   624→   624→          <CardContent className="p-4">
   625→   625→            <div className="flex items-center gap-2 mb-1">
   626→   626→              <div className={`w-7 h-7 rounded-lg ${totalPnl >= 0 ? 'bg-emerald-500/10' : 'bg-red-500/10'} flex items-center justify-center`}>
   627→   627→                <DollarSign className={`h-3.5 w-3.5 ${totalPnl >= 0 ? 'text-emerald-400' : 'text-red-400'}`} />
   628→   628→              </div>
   629→   629→              <span className="text-[10px] text-muted-foreground uppercase tracking-wider">Total P&L</span>
   630→   630→            </div>
