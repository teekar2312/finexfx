   113→// ---------- helpers ----------
   114→
   115→function getUtcHour(): number {
   116→  const now = new Date();
   117→  return now.getUTCHours() + now.getUTCMinutes() / 60 + now.getUTCSeconds() / 3600;
   118→}
   119→
   120→function isSessionActive(utcH: number, start: number, end: number): boolean {
   121→  if (start < end) return utcH >= start && utcH < end;
   122→  // wraps midnight (e.g. Sydney 22-7)
   123→  return utcH >= start || utcH < end;
   124→}
   125→
   126→function timeRemainingInSession(utcH: number, end: number): number {
   127→  if (utcH < end) return end - utcH;
   128→  return 24 - utcH + end;
   129→}
   130→
   131→function timeUntilSessionStart(utcH: number, start: number): number {
   132→  if (start > utcH) return start - utcH;
   133→  return 24 - utcH + start;
   134→}
   135→
   136→function formatCountdown(hours: number): string {
   137→  const h = Math.floor(hours);
   138→  const m = Math.floor((hours - h) * 60);
   139→  const s = Math.floor(((hours - h) * 60 - m) * 60);
   140→  return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
   141→}
   142→
   143→function getActiveSessions(utcH: number): SessionId[] {
   144→  return SESSIONS.filter(s => isSessionActive(utcH, s.start, s.end)).map(s => s.id as SessionId);
   145→}
   146→
   147→function getActiveOverlaps(utcH: number): OverlapInfo[] {
   148→  return OVERLAPS.filter(o => {
   149→    if (o.start < o.end) return utcH >= o.start && utcH < o.end;
   150→    return utcH >= o.start || utcH < o.end;
   151→  });
   152→}
   153→
   154→function getNextOverlap(utcH: number): OverlapInfo | null {
   155→  let best: OverlapInfo | null = null;
   156→  let bestDist = 999;
   157→  for (const o of OVERLAPS) {
   158→    const dist = timeUntilSessionStart(utcH, o.start);
   159→    if (dist < bestDist && dist > 0) {
   160→      bestDist = dist;
   161→      best = o;
   162→    }
   163→  }
   164→  return best;
   165→}
   166→
   167→function getTimelineBlocks(utcH: number): SessionBlock[] {
   168→  // render each session as a block on the 0-24 timeline
   169→  return SESSIONS.map(s => ({
   170→    id: s.id as SessionId,
   171→    label: s.label,
   172→    start: s.start,
   173→    end: s.end,
   174→    color: s.color,
   175→    text: s.text,
   176→    isActive: isSessionActive(utcH, s.start, s.end),
   177→  }));
   178→}
   179→
   180→function blockToPosition(start: number, end: number): { left: string; width: string } {
   181→  if (start < end) {
   182→    return { left: `${(start / 24) * 100}%`, width: `${((end - start) / 24) * 100}%` };
   183→  }
   184→  // wraps midnight: two segments
   185→  const w1 = 24 - start;
   186→  const w2 = end;
   187→  // just show the first segment for simplicity on the timeline
   188→  return { left: `${(start / 24) * 100}%`, width: `${(w1 / 24) * 100}%` };
   189→}
   190→
   191→function blockToPositionWrap(start: number, end: number): { left1: string; width1: string; left2: string; width2: string } | { left: string; width: string } {
   192→  if (start < end) {
   193→    return { left: `${(start / 24) * 100}%`, width: `${((end - start) / 24) * 100}%` };
   194→  }
   195→  const w1 = 24 - start;
   196→  return {
   197→    left1: `${(start / 24) * 100}%`,
   198→    width1: `${(w1 / 24) * 100}%`,
   199→    left2: '0%',
   200→    width2: `${(end / 24) * 100}%`,
   201→  };
   202→}
