    21→

    23→  const {

    25→    openTrades, closedTrades, addTrade, closeTrade, updateTrade,

    32→  const [oneClickMode, setOneClickMode] = useState(false);
    33→  const [confirmClose, setConfirmClose] = useState<string | null>(null);
    34→  const prevPriceRef = useRef<Record<string, number>>({});
    35→  const [priceFlash, setPriceFlash] = useState<Record<string, 'up' | 'down' | null>>({});
    36→
    37→  const price = prices[selectedSymbol];
    38→  const chartData = priceHistory[selectedSymbol] || [];
    39→

    42→      const prev = prevPriceRef.current[selectedSymbol];
    43→      if (prev !== undefined && price.bid !== prev) {
    44→        setPriceFlash(state => ({ ...state, [selectedSymbol]: price.bid > prev ? 'up' : 'down' }));
    45→        const timer = setTimeout(() => {
    46→          setPriceFlash(state => ({ ...state, [selectedSymbol]: null }));

    48→        return () => clearTimeout(timer);
    49→      }

    52→  }, [price, selectedSymbol]);

    56→      addNotification({ type: 'error', title: 'Cannot Trade', message: 'Not connected to price feed' });

    63→      return;
    64→    }
    65→
    66→    const entryPrice = direction === 'BUY' ? price.ask : price.bid;
    67→    const slPips = parseFloat(stopLoss) || 0;
    68→    const tpPips = parseFloat(takeProfit) || 0;
    69→    const pipSize = SYMBOL_INFO[selectedSymbol].pipSize;
    70→
    71→    const trade = {
    72→      id: `trade-${Date.now()}-${Math.random().toString(36).substr(2, 6)}`,
    73→      symbol: selectedSymbol,
    74→      direction,
    75→      lotSize: lots,
    76→      entryPrice,
    77→      currentPrice: entryPrice,
    78→      stopLoss: slPips > 0 ? (direction === 'BUY' ? entryPrice - slPips * pipSize : entryPrice + slPips * pipSize) : undefined,
    79→      takeProfit: tpPips > 0 ? (direction === 'BUY' ? entryPrice + tpPips * pipSize : entryPrice - tpPips * pipSize) : undefined,
    80→      isTrailingStop: false,
    81→      trailingStop: undefined,
    82→      pips: 0,
    83→      profit: 0,
    84→      commission: BROKER_CONFIG.commission * lots,
    85→      spread: price.spread,
    86→      swap: 0,
    87→      status: 'open' as const,
    88→      openedAt: new Date().toISOString(),

    91→    addTrade(trade);

    93→

    95→    updateTrade(tradeId, { isTrailingStop: enabled, trailingStop: enabled ? riskSettings.stopLossPips : undefined });
    96→  };
    97→
    98→  // (c) Calculated order metrics
    99→  const orderMetrics = useMemo(() => {

   101→    const slPips = parseFloat(stopLoss) || 0;
   102→    const tpPips = parseFloat(takeProfit) || 0;

   104→    const pipSize = SYMBOL_INFO[selectedSymbol].pipSize;
   105→
   106→    // Margin = (contractSize * lots * price) / leverage
   107→    const contractSize = SYMBOL_INFO[selectedSymbol].category === 'forex' ? 100000 : 100;
   108→    const currentPrice = price ? (selectedSymbol === 'USDJPY' ? 1 / price.bid : price.bid) : 1;
   109→    const margin = (contractSize * lots * currentPrice) / BROKER_CONFIG.leverage;
   110→
   111→    const riskAmount = slPips * pipValue;
   112→    const potentialProfit = tpPips * pipValue;
   113→
   114→    return { margin, riskAmount, potentialProfit, pipValue };
   115→  }, [lotSize, stopLoss, takeProfit, selectedSymbol, price]);
   116→

   119→    if (closedTrades.length === 0) return null;

   121→    const losses = closedTrades.filter(t => t.profit <= 0);
   122→    const winRate = (wins.length / closedTrades.length) * 100;
   123→    const totalPnl = closedTrades.reduce((s, t) => s + t.profit, 0);
   124→    const bestTrade = closedTrades.reduce((best, t) => t.profit > best.profit ? t : best, closedTrades[0]);
   125→    const worstTrade = closedTrades.reduce((worst, t) => t.profit < worst.profit ? t : worst, closedTrades[0]);
   126→    return { totalTrades: closedTrades.length, winRate, totalPnl, bestTrade, worstTrade };
   127→  }, [closedTrades]);
   128→
   129→  function formatDuration(openedAt: string, closedAt?: string) {

   133→    if (diffMin < 1) return '<1m';
   134→    if (diffMin < 60) return `${diffMin}m`;
   135→    const hr = Math.floor(diffMin / 60);
   136→    const min = diffMin % 60;
   137→    if (hr < 24) return `${hr}h ${min}m`;
   138→    const days = Math.floor(hr / 24);
   139→    const remHr = hr % 24;
   140→    return `${days}d ${remHr}h`;

   148→    const min = Math.min(...pts.map(p => p.bid));
   149→    const max = Math.max(...pts.map(p => p.bid));
   150→    const range = max - min || 1;
   151→    return pts.map((p, i) => `${(i / 11) * 36},${20 - ((p.bid - min) / range) * 18}`).join(' ');
   152→  }, [priceHistory]);
   153→
   154→  // Quick lot sizes
   155→  const quickLots = [0.01, 0.05, 0.1, 0.5, 1.0];
   156→
   157→  return (
   158→    <TooltipProvider delayDuration={0}>
   159→      <div className="p-4 pb-10 md:pb-4 space-y-4">

   163→            const isActive = sym === selectedSymbol;
   164→            const p = prices[sym];
   165→            const sparkline = getSparklinePoints(sym);
   166→            return (
   167→              <button
   168→                key={sym}
   169→                onClick={() => setSelectedSymbol(sym)}
   170→                className={`flex-shrink-0 py-2.5 px-3 rounded-full text-sm font-medium transition-all min-w-[120px] card-hover scale-click flex items-center gap-2 border ${
   171→                  isActive
   172→                    ? 'bg-emerald-500/10 text-emerald-400 border-emerald-500/30 pill-active-glow shadow-[0_0_20px_rgba(16,185,129,0.1)]'
   173→                    : 'bg-card/80 border-border/60 text-muted-foreground hover:text-foreground hover:border-border'
   174→                }`}
   175→              >
   176→                <div className="flex flex-col items-start gap-0.5">
   177→                  <div className="flex items-center gap-1.5">
   178→                    <span className="text-xs font-bold tracking-tight">{SYMBOL_INFO[sym].name}</span>
   179→                    {p && (
   180→                      <span className="text-[9px] text-muted-foreground font-normal tabular-nums">
   181→                        {p.spread.toFixed(1)}p
   182→                      </span>
   183→                    )}
   184→                  </div>
   185→                  {p && (

   187→                      {p.bid.toFixed(SYMBOL_INFO[sym].digits)}
   188→                    </div>
   189→                  )}
   190→                </div>
   191→                {sparkline && (
   192→                  <svg viewBox="0 0 36 20" className="w-9 h-5 flex-shrink-0" preserveAspectRatio="none">
   193→                    <polyline
   194→                      points={sparkline}
   195→                      fill="none"
   196→                      stroke={p && p.change >= 0 ? '#10b981' : '#ef4444'}
   197→                      strokeWidth="1.5"
   198→                      strokeLinecap="round"
   199→                      strokeLinejoin="round"
   200→                      opacity={isActive ? 1 : 0.6}
   201→                    />
   202→                  </svg>
   203→                )}
   204→              </button>
   205→            );
   206→          })}
   207→        </div>
   208→
   209→        <div className="grid grid-cols-1 lg:grid-cols-4 gap-4">
   210→          {/* Chart */}
   211→          <div className="lg:col-span-3 space-y-4">
   212→            <div className="chart-glow rounded-xl">
   213→              <Card className="glass-card border-border/40">
   214→                <CardContent className="p-0">
   215→                  <PriceChart
   216→                    data={chartData}
   217→                    symbol={selectedSymbol}
   218→                    bid={price?.bid}
   219→                    ask={price?.ask}
   220→                    height={300}
   221→                  />
   222→                </CardContent>
   223→              </Card>
   224→            </div>
   225→
   226→            {/* (a) Tabbed Positions / History */}
   227→            <Card className="glass-card card-hover">
   228→              <Tabs defaultValue="open" className="w-full">
   229→                <CardHeader className="pb-0 pt-3 px-4">
   230→                  <TabsList className="h-8 bg-slate-800/50">
   231→                    <TabsTrigger value="open" className="text-xs h-6 data-[state=active]:bg-emerald-600/20 data-[state=active]:text-emerald-400 data-[state=active]:shadow-[0_0_10px_rgba(16,185,129,0.15)]">
   232→                      Open ({openTrades.length})
   233→                    </TabsTrigger>
   234→                    <TabsTrigger value="history" className="text-xs h-6 data-[state=active]:bg-emerald-600/20 data-[state=active]:text-emerald-400 data-[state=active]:shadow-[0_0_10px_rgba(16,185,129,0.15)]">
   235→                      History ({closedTrades.length})

   238→                </CardHeader>
   239→
   240→                <TabsContent value="open" className="mt-0">
   241→                  <CardContent className="px-4 pb-3">
   242→                    {openTrades.length === 0 ? (
   243→                      <div className="text-center py-6 text-muted-foreground text-sm">
   244→                        No open positions. Use the panel to place a trade.

   246→                    ) : (
   247→                      <div className="overflow-x-auto">
   248→                        <Table>
   249→                          <TableHeader>
   250→                            <TableRow className="border-border hover:bg-transparent">
   251→                              <TableHead className="text-[10px] h-8">Symbol</TableHead>
   252→                              <TableHead className="text-[10px] h-8">Dir</TableHead>
   253→                              <TableHead className="text-[10px] h-8 text-right">Lots</TableHead>

   256→                              <TableHead className="text-[10px] h-8 text-right">TP</TableHead>
   257→                              <TableHead className="text-[10px] h-8 text-right">Pips</TableHead>
   258→                              <TableHead className="text-[10px] h-8 text-right">P&L</TableHead>
   259→                              <TableHead className="text-[10px] h-8">Time</TableHead>

   264→                          <TableBody>
   265→                            {openTrades.map((trade) => {
   266→                              const maxProfit = Math.max(...openTrades.map(t => Math.abs(t.profit)), 1);
   267→                              return (
   268→                              <TableRow key={trade.id} className={`border-border compact-row ${trade.direction === 'BUY' ? 'trade-row-buy' : 'trade-row-sell'}`}>
   269→                                <TableCell className="text-xs font-medium">{trade.symbol}</TableCell>
   270→                                <TableCell>
   271→                                  <div className={`flex items-center gap-1 text-xs font-bold ${trade.direction === 'BUY' ? 'text-emerald-500' : 'text-red-500'}`}>
   272→                                    {trade.direction === 'BUY' ? <ArrowUpRight className="h-3 w-3" /> : <ArrowDownRight className="h-3 w-3" />}
   273→                                    {trade.direction}
   274→                                  </div>
   275→                                </TableCell>
   276→                                <TableCell className="text-xs text-right tabular-nums">{trade.lotSize}</TableCell>
   277→                                <TableCell className="text-xs text-right tabular-nums">{trade.entryPrice.toFixed(SYMBOL_INFO[trade.symbol].digits)}</TableCell>
   278→                                <TableCell className="text-xs text-right tabular-nums text-red-400">
   279→                                  {trade.stopLoss?.toFixed(SYMBOL_INFO[trade.symbol].digits) || '—'}
   280→                                </TableCell>
   281→                                <TableCell className="text-xs text-right tabular-nums text-emerald-400">
   282→                                  {trade.takeProfit?.toFixed(SYMBOL_INFO[trade.symbol].digits) || '—'}
   283→                                </TableCell>
   284→                                <TableCell className={`text-xs text-right tabular-nums font-semibold ${trade.pips >= 0 ? 'text-emerald-500' : 'text-red-500'}`}>
   285→                                  {trade.pips >= 0 ? '+' : ''}{trade.pips.toFixed(1)}
   286→                                </TableCell>
   287→                                <TableCell className="text-right">
   288→                                  <div className={`text-sm font-bold tabular-nums ${trade.profit >= 0 ? 'gradient-text-profit' : 'gradient-text-loss'}`}>
   289→                                    {trade.profit >= 0 ? '+' : ''}${trade.profit.toFixed(2)}
   290→                                  </div>
   291→                                  <div className="pnl-bar-track mt-0.5">
   292→                                    <div
   293→                                      className={`pnl-bar-fill ${trade.profit >= 0 ? 'bg-emerald-500' : 'bg-red-500'}`}
   294→                                      style={{ width: `${Math.min((Math.abs(trade.profit) / maxProfit) * 100, 100)}%` }}
   295→                                    />
   296→                                  </div>
   297→                                </TableCell>
   298→                                <TableCell className="text-[10px] text-muted-foreground tabular-nums whitespace-nowrap">
   299→                                  <div className="flex items-center gap-1">
   300→                                    <Timer className="h-2.5 w-2.5 opacity-50" />
   301→                                    {formatDuration(trade.openedAt)}
   302→                                  </div>
   303→                                </TableCell>
   304→                                <TableCell className="text-center">
   305→                                  {trade.isTrailingStop ? (
   306→                                    <Tooltip>
   307→                                      <TooltipTrigger>
   308→                                        <div className="trail-animate inline-flex">
   309→                                          <Activity className="h-3.5 w-3.5 text-emerald-500" />
   310→                                        </div>
   311→                                      </TooltipTrigger>
   312→                                      <TooltipContent className="text-[10px]">Trailing {trade.trailingStop} pips</TooltipContent>
   313→                                    </Tooltip>
   314→                                  ) : (
   315→                                    <Switch
   316→                                      checked={trade.isTrailingStop}
   317→                                      onCheckedChange={(checked) => handleToggleTrailing(trade.id, checked)}

   319→                                    />
   320→                                  )}
   321→                                </TableCell>
   322→                                <TableCell className="text-center">
   323→                                  <AnimatePresence>
   324→                                    {confirmClose === trade.id ? (
   325→                                      <motion.div
   326→                                        initial={{ opacity: 0, scale: 0.8 }}
   327→                                        animate={{ opacity: 1, scale: 1 }}
   328→                                        exit={{ opacity: 0, scale: 0.8 }}
   329→                                        className="flex items-center gap-1"
   330→                                      >
   331→                                        <Button
   332→                                          size="sm"
   333→                                          variant="destructive"
   334→                                          className="h-6 px-2 text-[10px]"
   335→                                          onClick={() => { closeTrade(trade.id); setConfirmClose(null); }}
   336→                                        >
   337→                                          Confirm
   338→                                        </Button>
   339→                                        <Button
   340→                                          size="sm"
   341→                                          variant="ghost"
   342→                                          className="h-6 px-2 text-[10px]"
   343→                                          onClick={() => setConfirmClose(null)}
   344→                                        >
   345→                                          <X className="h-3 w-3" />
   346→                                        </Button>
   347→                                      </motion.div>
   348→                                    ) : (
   349→                                      <Button
   350→                                        size="sm"
   351→                                        variant="ghost"
   352→                                        className="h-6 px-2 text-muted-foreground hover:text-red-500 hover:bg-red-500/15 transition-colors duration-200"
   353→                                        onClick={() => setConfirmClose(trade.id)}
   354→                                      >
   355→                                        <Square className="h-3 w-3" />
   356→                                      </Button>
   357→                                    )}
   358→                                  </AnimatePresence>
   359→                                </TableCell>
   360→                              </TableRow>
   361→                              );
   362→                            })}
   363→                          </TableBody>
   364→                        </Table>
   365→                      </div>
   366→                    )}
   367→                  </CardContent>
   368→                </TabsContent>
   369→
   370→                <TabsContent value="history" className="mt-0">
   371→                  <CardContent className="px-4 pb-3">
   372→                    {closedTrades.length === 0 ? (
   373→                      <div className="text-center py-10 text-muted-foreground">
   374→                        <History className="h-8 w-8 mx-auto mb-2 opacity-30" />

   379→                      <>
   380→                        {/* Summary Stats + Win/Loss Ratio Bar + Avg Comparison */}
   381→                        {historyStats && (() => {
   382→                          const wins = closedTrades.filter(t => t.profit > 0);
   383→                          const losses = closedTrades.filter(t => t.profit <= 0);

   386→                          const maxAvg = Math.max(avgWin, avgLoss, 1);
   387→                          return (
   388→                            <>
   389→                            <div className="grid grid-cols-5 gap-2 mb-3">

   394→                              <div className="text-center p-2 rounded-lg bg-slate-800/30 stat-card-pattern">
   395→                                <div className="text-[10px] text-muted-foreground uppercase">Win Rate</div>
   396→                                <div className={`text-sm font-bold tabular-nums ${historyStats.winRate >= 50 ? 'text-emerald-500' : 'text-red-500'}`}>
   397→                                  {historyStats.winRate.toFixed(0)}%
