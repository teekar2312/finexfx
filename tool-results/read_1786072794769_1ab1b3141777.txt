    61→function ImpactDots({ impact }: { impact: 'high' | 'medium' | 'low' }) {
    62→  const count = impact === 'high' ? 3 : impact === 'medium' ? 2 : 1;
    63→  const color = impact === 'high' ? 'bg-red-500' : impact === 'medium' ? 'bg-amber-500' : 'bg-green-500';
    64→  return (
    65→    <div className="flex items-center gap-0.5">
    66→      {Array.from({ length: 3 }).map((_, i) => (
    67→        <div key={i} className={`w-1.5 h-1.5 rounded-full ${i < count ? color : 'bg-slate-700'}`} />
    68→      ))}
    69→    </div>
    70→  );
    71→}

    73→function getActualVsForecast(actual: string, forecast: string): 'better' | 'worse' | 'same' {
    74→  const a = parseFloat(actual.replace('%', '').replace('K', ''));
    75→  const f = parseFloat(forecast.replace('%', '').replace('K', ''));
    76→  if (isNaN(a) || isNaN(f)) return 'same';
    77→  if (a > f) return 'better';
    78→  if (a < f) return 'worse';
    79→  return 'same';
    80→}
    81→
    82→function getActualColor(actual: string, forecast: string): string {
    83→  const comparison = getActualVsForecast(actual, forecast);
    84→  return comparison === 'better' ? 'text-emerald-500' : comparison === 'worse' ? 'text-red-500' : 'text-foreground';
    85→}
    86→
    87→function ActualVsForecastArrow({ actual, forecast }: { actual: string; forecast: string }) {
    88→  const comparison = getActualVsForecast(actual, forecast);
    89→  if (comparison === 'better') return <ArrowUpRight className="h-3 w-3 text-emerald-500" />;
    90→  if (comparison === 'worse') return <ArrowDownRight className="h-3 w-3 text-red-500" />;
    91→  return <Minus className="h-3 w-3 text-slate-500" />;
    92→}

    94→function timeAgo(dateStr: string): string {
    95→  const now = Date.now();
    96→  const then = new Date(dateStr).getTime();
    97→  const diff = Math.floor((now - then) / 1000);
    98→  if (diff < 60) return `${diff}s ago`;
    99→  if (diff < 3600) return `${Math.floor(diff / 60)}m ago`;
   100→  if (diff < 86400) return `${Math.floor(diff / 3600)}h ago`;
   101→  return `${Math.floor(diff / 86400)}d ago`;
   102→}
