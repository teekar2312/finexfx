    17→function DonutGauge({ value, max, size = 80, strokeWidth = 6, showPercent = false }: { value: number; max: number; size?: number; strokeWidth?: number; showPercent?: boolean }) {
    18→  const radius = (size - strokeWidth) / 2;
    19→  const circumference = 2 * Math.PI * radius;
    20→  const ratio = Math.min(value / max, 1);
    21→  const offset = circumference * (1 - ratio);
    22→  const isSafe = ratio < 0.5;
    23→  const isWarn = ratio >= 0.5 && ratio < 0.8;
    24→  const isDanger = ratio >= 0.8;
    25→  const color = isSafe ? '#10b981' : isWarn ? '#f59e0b' : '#ef4444';
    26→  const glowColor = isSafe ? 'rgba(16,185,129,0.35)' : isWarn ? 'rgba(245,158,11,0.35)' : 'rgba(239,68,68,0.35)';
    27→  const pct = Math.round(ratio * 100);
    28→
    29→  return (
    30→    <motion.div
    31→      initial={{ opacity: 0, scale: 0.85 }}
    32→      animate={{ opacity: 1, scale: 1 }}
    33→      transition={{ duration: 0.4, ease: 'easeOut' }}
    34→      className="relative inline-flex items-center justify-center"
    35→    >
    36→      <svg width={size} height={size} className="-rotate-90">
    37→        <defs>
    38→          <filter id={`gauge-glow-${size}`}>
    39→            <feGaussianBlur stdDeviation="3" result="blur" />
    40→            <feMerge><feMergeNode in="blur" /><feMergeNode in="SourceGraphic" /></feMerge>
    41→          </filter>
    42→        </defs>
    43→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="rgba(255,255,255,0.06)" strokeWidth={strokeWidth} />
    44→        {/* Glow layer */}
    45→        <circle
    46→          cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color} strokeWidth={strokeWidth + 4}
    47→          strokeDasharray={circumference} strokeDashoffset={offset} strokeLinecap="round"
    48→          style={{ transition: 'stroke-dashoffset 0.6s ease, stroke 0.3s', opacity: 0.2, filter: `drop-shadow(0 0 6px ${glowColor})` }}
    49→        />
    50→        {/* Main arc */}
    51→        <circle
    52→          cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color} strokeWidth={strokeWidth}
    53→          strokeDasharray={circumference} strokeDashoffset={offset} strokeLinecap="round"
    54→          style={{ transition: 'stroke-dashoffset 0.6s ease, stroke 0.3s' }}
    55→          filter={`url(#gauge-glow-${size})`}
    56→        />
    57→      </svg>
    58→      {showPercent && (
    59→        <div className="absolute inset-0 flex flex-col items-center justify-center rotate-0">
    60→          <span className={`text-sm font-bold tabular-nums ${isSafe ? 'text-emerald-400' : isWarn ? 'text-amber-400' : 'text-red-400'}`}>{pct}%</span>
    61→        </div>
    62→      )}
    63→    </motion.div>
    64→  );
    65→}
