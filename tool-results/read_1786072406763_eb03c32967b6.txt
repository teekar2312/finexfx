   115→// ─── Pattern Detection ────────────────────────────────────────────────────────
   116→
   117→function detectPatterns(candles: PriceHistory[]): DetectedPattern[] {
   118→  const patterns: DetectedPattern[] = [];
   119→
   120→  for (let i = 0; i < candles.length; i++) {
   121→    const c = candles[i];
   122→    const bodySize = Math.abs(c.close - c.open);
   123→    const upperWick = c.high - Math.max(c.close, c.open);
   124→    const lowerWick = Math.min(c.close, c.open) - c.low;
   125→    const totalRange = c.high - c.low;
   126→
   127→    // ── Doji ──
   128→    if (totalRange > 0 && bodySize / totalRange < 0.15 && totalRange > 0.0004) {
   129→      patterns.push({
   130→        name: 'Doji',
   131→        type: 'neutral',
   132→        reliability: 'Medium',
   133→        description: 'Open and close are nearly equal, indicating market indecision and potential reversal.',
   134→        startIndex: i,
   135→        endIndex: i,
   136→      });
   137→    }
   138→    // ── Hammer ──
   139→    else if (
   140→      totalRange > 0.0006 &&
   141→      lowerWick > bodySize * 2 &&
   142→      upperWick < bodySize * 0.5 &&
   143→      bodySize < totalRange * 0.35
   144→    ) {
   145→      patterns.push({
   146→        name: 'Hammer',
   147→        type: 'bullish',
   148→        reliability: 'High',
   149→        description: 'Small body at top with long lower shadow. Buyers rejected lower prices, signaling potential bullish reversal.',
   150→        startIndex: i,
   151→        endIndex: i,
   152→      });
   153→    }
   154→    // ── Inverted Hammer ──
   155→    else if (
   156→      totalRange > 0.0006 &&
   157→      upperWick > bodySize * 2 &&
   158→      lowerWick < bodySize * 0.5 &&
   159→      bodySize < totalRange * 0.35
   160→    ) {
   161→      patterns.push({
   162→        name: 'Inverted Hammer',
   163→        type: 'bullish',
   164→        reliability: 'Medium',
   165→        description: 'Small body at bottom with long upper shadow. Shows buying pressure despite selling, suggesting reversal potential.',
   166→        startIndex: i,
   167→        endIndex: i,
   168→      });
   169→    }
   170→    // ── Spinning Top ──
   171→    else if (
   172→      totalRange > 0.0008 &&
   173→      bodySize < totalRange * 0.25 &&
   174→      upperWick > bodySize * 1.5 &&
   175→      lowerWick > bodySize * 1.5
   176→    ) {
   177→      patterns.push({
   178→        name: 'Spinning Top',
   179→        type: 'neutral',
   180→        reliability: 'Low',
   181→        description: 'Small body with roughly equal upper and lower wicks. Represents market indecision between buyers and sellers.',
   182→        startIndex: i,
   183→        endIndex: i,
   184→      });
   185→    }
   186→
   187→    // ── Bullish Engulfing (2-candle pattern) ──
   188→    if (i >= 1) {
   189→      const prev = candles[i - 1];
   190→      const prevBody = Math.abs(prev.close - prev.open);
   191→      const prevBearish = prev.close < prev.open;
   192→      const currBullish = c.close > c.open;
   193→
   194→      if (
   195→        prevBearish &&
   196→        currBullish &&
   197→        bodySize > prevBody * 1.1 &&
   198→        c.open <= prev.close &&
   199→        c.close >= prev.open
   200→      ) {
   201→        patterns.push({
   202→          name: 'Bullish Engulfing',
   203→          type: 'bullish',
   204→          reliability: 'High',
   205→          description: 'A large green candle completely engulfs the previous red candle, signaling strong bullish reversal momentum.',
   206→          startIndex: i - 1,
   207→          endIndex: i,
   208→        });
   209→      }
   210→
   211→      // ── Bearish Engulfing (2-candle pattern) ──
   212→      const prevBullish = prev.close > prev.open;
   213→      const currBearish = c.close < c.open;
   214→
   215→      if (
   216→        prevBullish &&
   217→        currBearish &&
   218→        bodySize > prevBody * 1.1 &&
   219→        c.open >= prev.close &&
   220→        c.close <= prev.open
   221→      ) {
   222→        patterns.push({
   223→          name: 'Bearish Engulfing',
   224→          type: 'bearish',
   225→          reliability: 'High',
   226→          description: 'A large red candle completely engulfs the previous green candle, indicating strong bearish reversal pressure.',
   227→          startIndex: i - 1,
   228→          endIndex: i,
   229→        });
   230→      }
   231→    }
   232→
   233→    // ── Morning Star (3-candle pattern) ──
   234→    if (i >= 2) {
   235→      const c1 = candles[i - 2];
   236→      const c2 = candles[i - 1];
   237→      const prevBody1 = Math.abs(c1.close - c1.open);
   238→      const starBody = Math.abs(c2.close - c2.open);
   239→      const c1Bearish = c1.close < c1.open;
   240→
   241→      if (
   242→        c1Bearish &&
   243→        prevBody1 > 0.0008 &&
   244→        starBody < prevBody1 * 0.35 &&
   245→        c.close > c1.open &&
   246→        bodySize > prevBody1 * 0.7 &&
   247→        c.close > c.open
   248→      ) {
   249→        patterns.push({
   250→          name: 'Morning Star',
   251→          type: 'bullish',
   252→          reliability: 'High',
   253→          description: 'Three-candle reversal: large red, small indecision body, large green. Signals transition from downtrend to uptrend.',
   254→          startIndex: i - 2,
   255→          endIndex: i,
   256→        });
   257→      }
   258→
   259→      // ── Evening Star (3-candle pattern) ──
   260→      const prevBody = Math.abs(c1.close - c1.open);
   261→      const c1Bullish = c1.close > c1.open;
   262→
   263→      if (
   264→        c1Bullish &&
   265→        prevBody > 0.0008 &&
   266→        starBody < prevBody * 0.35 &&
   267→        c.close < c1.open &&
   268→        bodySize > prevBody * 0.7 &&
   269→        c.close < c.open
   270→      ) {
   271→        patterns.push({
   272→          name: 'Evening Star',
   273→          type: 'bearish',
   274→          reliability: 'High',
   275→          description: 'Three-candle reversal: large green, small indecision body, large red. Signals transition from uptrend to downtrend.',
   276→          startIndex: i - 2,
   277→          endIndex: i,
   278→        });
   279→      }
   280→    }
   281→  }
   282→
   283→  return patterns;
   284→}

   289→  const d = new Date(ts);
   290→  return `${d.getMonth() + 1}/${d.getDate()} ${String(d.getHours()).padStart(2, '0')}:00`;

   344→// ─── SVG Candlestick Chart ───────────────────────────────────────────────────
   345→
   346→interface ChartProps {
   347→  candles: PriceHistory[];
   348→  patterns: DetectedPattern[];
   349→  chartWidth: number;
   350→  chartHeight: number;
   351→  priceTop: number;
   352→  priceBottom: number;
   353→}
   354→
   355→function CandlestickChartSVG({
   356→  candles,
   357→  patterns,
   358→  chartWidth,
   359→  chartHeight,
   360→  priceTop,
   361→  priceBottom,
   362→}: ChartProps) {
   363→  const padding = { top: 40, right: 12, bottom: 32, left: 56 };
   364→  const plotWidth = chartWidth - padding.left - padding.right;
   365→  const plotHeight = chartHeight - padding.top - padding.bottom;
   366→  const candleWidth = Math.max(3, plotWidth / candles.length * 0.6);
   367→  const candleSpacing = plotWidth / candles.length;
   368→
   369→  const yScale = (price: number) =>
   370→    padding.top + plotHeight - ((price - priceBottom) / (priceTop - priceBottom)) * plotHeight;
   371→
   372→  const xScale = (index: number) =>
   373→    padding.left + candleSpacing * index + candleSpacing / 2;
   374→
   375→  // Price grid lines
   376→  const priceRange = priceTop - priceBottom;
   377→  const gridStep = priceRange > 0.01 ? 0.002 : priceRange > 0.005 ? 0.001 : 0.0005;
   378→  const gridPrices: number[] = [];
   379→  for (let p = Math.ceil(priceBottom / gridStep) * gridStep; p <= priceTop; p += gridStep) {
   380→    gridPrices.push(parseFloat(p.toFixed(5)));
   381→  }
   382→
   383→  // Build highlighted zones for patterns
   384→  const highlightedZones = patterns.map((pat) => {
   385→    const x1 = xScale(pat.startIndex) - candleSpacing / 2 - 2;
   386→    const x2 = xScale(pat.endIndex) + candleSpacing / 2 + 2;
   387→    const allHighs = candles.slice(pat.startIndex, pat.endIndex + 1).map((c) => c.high);
   388→    const allLows = candles.slice(pat.startIndex, pat.endIndex + 1).map((c) => c.low);
   389→    const zoneTop = yScale(Math.max(...allHighs)) - 4;
   390→    const zoneBottom = yScale(Math.min(...allLows)) + 4;
   391→    const midX = (x1 + x2) / 2;
   392→    const labelY = zoneTop - 6;
   393→    return { ...pat, x1, x2, zoneTop, zoneBottom, midX, labelY };
   394→  });
   395→
   396→  return (
   397→    <svg
   398→      width="100%"
   399→      height="100%"
   400→      viewBox={`0 0 ${chartWidth} ${chartHeight}`}
   401→      className="overflow-visible"
   402→      preserveAspectRatio="xMidYMid meet"
   403→    >
   404→      <defs>
   405→        <linearGradient id="chartBgGrad" x1="0" y1="0" x2="0" y2="1">
   406→          <stop offset="0%" stopColor="rgba(17, 24, 39, 0.4)" />
   407→          <stop offset="100%" stopColor="rgba(17, 24, 39, 0.8)" />
   408→        </linearGradient>
   409→      </defs>
   410→
   411→      {/* Background */}
   412→      <rect x="0" y="0" width={chartWidth} height={chartHeight} rx="8" fill="url(#chartBgGrad)" />
   413→
   414→      {/* Price grid lines */}
   415→      {gridPrices.map((price) => (
   416→        <g key={price}>
   417→          <line
   418→            x1={padding.left}
   419→            y1={yScale(price)}
   420→            x2={chartWidth - padding.right}
   421→            y2={yScale(price)}
   422→            stroke="rgba(255,255,255,0.04)"
   423→            strokeWidth={1}
   424→          />
   425→          <text
   426→            x={padding.left - 6}
   427→            y={yScale(price) + 3}
   428→            textAnchor="end"
   429→            fill="rgba(148,163,184,0.6)"
   430→            fontSize={9}
   431→            fontFamily="monospace"
   432→          >
   433→            {price.toFixed(4)}
   434→          </text>
   435→        </g>
   436→      ))}
   437→
   438→      {/* Highlighted pattern zones (behind candles) */}
   439→      {highlightedZones.map((zone, idx) => (
   440→        <g key={`zone-${idx}`}>
   441→          <rect
   442→            x={zone.x1}
   443→            y={zone.zoneTop + 12}
   444→            width={zone.x2 - zone.x1}
   445→            height={zone.zoneBottom - zone.zoneTop - 12}
   446→            rx={4}
   447→            fill={patternZoneColor(zone.type)}
   448→            stroke={patternBorderColor(zone.type)}
   449→            strokeWidth={1.2}
   450→            strokeDasharray="4 2"
   451→          />
   452→          <text
   453→            x={zone.midX}
   454→            y={zone.labelY}
   455→            textAnchor="middle"
   456→            fill={patternBorderColor(zone.type)}
   457→            fontSize={8}
   458→            fontWeight={600}
   459→            fontFamily="system-ui, sans-serif"
   460→          >
   461→            {zone.name}
   462→          </text>
   463→        </g>
   464→      ))}
   465→
   466→      {/* Candles */}
   467→      {candles.map((candle, idx) => {
   468→        const isBullish = candle.close > candle.open;
   469→        const bodyTop = yScale(Math.max(candle.close, candle.open));
   470→        const bodyBottom = yScale(Math.min(candle.close, candle.open));
   471→        const bodyHeight = Math.max(1, bodyBottom - bodyTop);
   472→        const cx = xScale(idx);
   473→        const color = isBullish ? '#10b981' : '#ef4444';
   474→        const fillColor = isBullish ? 'rgba(16, 185, 129, 0.9)' : 'rgba(239, 68, 68, 0.9)';
   475→
   476→        return (
   477→          <g key={`candle-${idx}`}>
   478→            {/* Upper wick */}
   479→            <line
   480→              x1={cx}
   481→              y1={yScale(candle.high)}
   482→              x2={cx}
   483→              y2={bodyTop}
   484→              stroke={color}
   485→              strokeWidth={1}
   486→            />
   487→            {/* Lower wick */}
   488→            <line
   489→              x1={cx}
   490→              y1={bodyBottom}
   491→              x2={cx}
   492→              y2={yScale(candle.low)}
   493→              stroke={color}
   494→              strokeWidth={1}
   495→            />
   496→            {/* Body */}
   497→            <rect
   498→              x={cx - candleWidth / 2}
   499→              y={bodyTop}
   500→              width={candleWidth}
   501→              height={bodyHeight}
   502→              rx={1}
   503→              fill={fillColor}
   504→              stroke={color}
   505→              strokeWidth={0.5}
   506→            />
   507→          </g>
   508→        );
   509→      })}
   510→
   511→      {/* X-axis labels */}
   512→      {candles
   513→        .filter((_, i) => i % 4 === 0)
   514→        .map((candle, displayIdx) => {
